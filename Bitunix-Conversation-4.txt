        canvases.some(c => /chart/i.test(c.ariaLabel));

      frames.push({
        path: item.path,
        accessible: true,
        url,
        title,
        isChart,
        canvasCount,
        chartMarkers,
        canvases,
        win,
        doc
      });

      if (item.depth >= 3) continue;

      let childCount = 0;

      try {
        childCount = win.frames.length;
      } catch (_) {}

      for (let i = 0; i < childCount; i++) {
        try {
          queue.push({
            win: win.frames[i],
            path: item.path + " > iframe[" + i + "]",
            depth: item.depth + 1
          });
        } catch (_) {}
      }
    }

    return frames;
  }

  // ------------------------------------------------------------
  // Analyze a function and calculate relevance score.
  // ------------------------------------------------------------

  function analyzeFunction(fn, path) {
    const source = safeFunctionSource(fn);
    const name = cleanText(
      getSafeProperty(fn, "name") || "(anonymous)",
      100
    );

    if (!source) return null;

    let score = 0;
    const matches = [];

    if (/close.?position|position.?close|market.?close/i.test(path)) {
      score += 7;
      matches.push("path-close-position");
    }

    if (/reverse.?position|position.?reverse/i.test(path)) {
      score += 7;
      matches.push("path-reverse-position");
    }

    if (/reduce.?only/i.test(path)) {
      score += 7;
      matches.push("path-reduce-only");
    }

    if (/close/i.test(path)) {
      score += 4;
      matches.push("path-close");
    }

    if (/reverse/i.test(path)) {
      score += 4;
      matches.push("path-reverse");
    }

    if (/position/i.test(path)) {
      score += 1;
    }

    if (/order|trade/i.test(path)) {
      score += 1;
    }

    const sourceTests = [
      {
        re: /close[\w$.-]{0,35}position|position[\w$.-]{0,35}close/i,
        weight: 6,
        tag: "source-close-position"
      },
      {
        re: /market[\w$.-]{0,25}close/i,
        weight: 5,
        tag: "source-market-close"
      },
      {
        re: /reverse[\w$.-]{0,35}position|position[\w$.-]{0,35}reverse/i,
        weight: 6,
        tag: "source-reverse-position"
      },
      {
        re: /reduce[\w$.-]{0,15}only/i,
        weight: 6,
        tag: "source-reduce-only"
      },
      {
        re: /close[\w$.-]{0,20}order/i,
        weight: 4,
        tag: "source-close-order"
      },
      {
        re: /reverse[\w$.-]{0,20}order/i,
        weight: 4,
        tag: "source-reverse-order"
      },
      {
        re: /close/i,
        weight: 1,
        tag: "source-close"
      },
      {
        re: /reverse/i,
        weight: 2,
        tag: "source-reverse"
      }
    ];

    for (const test of sourceTests) {
      if (test.re.test(source)) {
        score += test.weight;
        matches.push(test.tag);
      }
    }

    if (score < 3) return null;

    const snippets = [];

    for (const pattern of SOURCE_PATTERNS) {
      pattern.lastIndex = 0;
      const match = pattern.exec(source);

      if (match) {
        const start = Math.max(0, match.index - 180);
        const end = Math.min(
          source.length,
          match.index + match[0].length + 260
        );

        const snippet = cleanText(source.slice(start, end), 550);

        if (snippet && !snippets.includes(snippet)) {
          snippets.push(snippet);
        }
      }
    }

    return {
      score,
      path: cleanText(path, 350),
      functionName: name,
      functionLength: source.length,
      matches: [...new Set(matches)],
      sourceSnippets: snippets.slice(0, 5)
    };
  }

  // ------------------------------------------------------------
  // Scan JavaScript object trees without invoking their methods.
  // ------------------------------------------------------------

  function scanObjectRoots(win, doc, framePath) {
    const functionResults = [];
    const objectResults = [];
    const rootNames = [];
    const seenObjects = new WeakSet();
    const seenFunctions = new WeakSet();
    const queue = [];

    let inspectedObjects = 0;
    let inspectedFunctions = 0;

    const MAX_OBJECTS = 4200;
    const MAX_FUNCTIONS = 9000;
    const MAX_DEPTH = 5;
    const MAX_KEYS_PER_OBJECT = 100;

    function addQueue(value, path, depth) {
      if (!value || depth > MAX_DEPTH) return;

      if (
        typeof value !== "object" &&
        typeof value !== "function"
      ) return;

      if (isDOMObject(value, win)) return;

      if (typeof value === "object") {
        try {
          if (value === win || value === doc) return;
        } catch (_) {}
      }

      queue.push({ value, path, depth });
    }

    function addFunction(fn, path) {
      if (typeof fn !== "function") return;
      if (seenFunctions.has(fn)) return;

      seenFunctions.add(fn);
      inspectedFunctions++;

      const result = analyzeFunction(fn, path);

      if (result) {
        functionResults.push({
          frame: framePath,
          ...result
        });
      }
    }

    // Find relevant global roots by name.
    let globalNames = [];

    try {
      globalNames = Object.getOwnPropertyNames(win);
    } catch (_) {}

    const selectedNames = globalNames.filter(name =>
      ROOT_HINTS.test(name) &&
      !SENSITIVE_KEY.test(name)
    );

    const knownNames = [
      "tradingViewApi",
      "ChartApiInstance",
      "chartWidget",
      "chartWidgetCollection",
      "chartWidgetCollectionApi",
      "tradingView",
      "widget",
      "store"
    ];

    for (const name of knownNames) {
      if (globalNames.includes(name) && !selectedNames.includes(name)) {
        selectedNames.push(name);
      }
    }

    for (const name of selectedNames.slice(0, 180)) {
      const value = getOwnValue(win, name);

      if (
        value === undefined ||
        SENSITIVE_KEY.test(name)
      ) continue;

      rootNames.push(name);
      addQueue(value, "window." + name, 0);

      if (typeof value === "function") {
        addFunction(value, "window." + name);
      }
    }

    // Inspect selected roots and their descendants.
    while (queue.length && inspectedObjects < MAX_OBJECTS) {
      const item = queue.shift();
      const obj = item.value;

      if (!obj) continue;

      if (typeof obj === "object") {
        if (seenObjects.has(obj)) continue;
        seenObjects.add(obj);
      } else if (typeof obj === "function") {
        addFunction(obj, item.path);
        if (seenObjects.has(obj)) continue;
        seenObjects.add(obj);
      }

      inspectedObjects++;

      let keys = [];

      try {
        keys = Object.getOwnPropertyNames(obj).slice(
          0,
          MAX_KEYS_PER_OBJECT
        );
      } catch (_) {
        continue;
      }

      const relevantKeys = [];

      for (const key of keys) {
        if (SENSITIVE_KEY.test(key)) continue;

        const descriptor = (() => {
          try {
            return Object.getOwnPropertyDescriptor(obj, key);
          } catch (_) {
            return null;
          }
        })();

        // Accessor properties are deliberately not invoked.
        if (!descriptor ||
            !Object.prototype.hasOwnProperty.call(descriptor, "value")) {
          continue;
        }

        const value = descriptor.value;
        const childPath = item.path + "." + key;

        if (typeof value === "function") {
          if (inspectedFunctions < MAX_FUNCTIONS) {
            addFunction(value, childPath);
          }

          if (FUNCTION_HINTS.test(key)) {
            relevantKeys.push({
              key,
              type: "function"
            });
          }

          // Do not recursively explore arbitrary function internals.
          continue;
        }

        if (
          value &&
          typeof value === "object" &&
          !isDOMObject(value, win)
        ) {
          if (FUNCTION_HINTS.test(key)) {
            relevantKeys.push({
              key,
              type: Array.isArray(value) ? "array" : "object"
            });
          }

          if (Array.isArray(value)) {
            // Limited inspection of module arrays.
            for (let i = 0; i < Math.min(value.length, 45); i++) {
              const entry = getOwnValue(value, String(i));

              if (
                entry &&
                (typeof entry === "object" ||
                 typeof entry === "function")
              ) {
                addQueue(
                  entry,
                  childPath + "[" + i + "]",
                  item.depth + 1
                );
              }
            }
          } else {
            addQueue(value, childPath, item.depth + 1);
          }
        }
      }

      if (relevantKeys.length) {
        objectResults.push({
          frame: framePath,
          path: cleanText(item.path, 350),
          matchingKeys: relevantKeys.slice(0, 60)
        });
      }

      // Inspect a small number of prototype levels.
      if (item.depth <= 2) {
        let proto = null;

        try {
          proto = Object.getPrototypeOf(obj);
        } catch (_) {}

        for (let level = 0; proto && level < 2; level++) {
          let protoKeys = [];

          try {
            protoKeys = Object.getOwnPropertyNames(proto);
          } catch (_) {}

          for (const key of protoKeys) {
            if (SENSITIVE_KEY.test(key)) continue;

            const value = getOwnValue(proto, key);

            if (typeof value === "function") {
              if (FUNCTION_HINTS.test(key)) {
                addFunction(
                  value,
                  item.path + ".[[Prototype]]." + key
                );
              }
            }
          }

          try {
            proto = Object.getPrototypeOf(proto);
          } catch (_) {
            break;
          }
        }
      }
    }

    functionResults.sort((a, b) => b.score - a.score);

    return {
      globalRootCandidates: rootNames,
      inspectedObjects,
      inspectedFunctions,
      objectMatches: objectResults.slice(0, 180),
      functionMatches: functionResults.slice(0, 160)
    };
  }

  // ------------------------------------------------------------
  // Inspect React props/fibers attached to relevant DOM nodes.
  // No handlers are called.
  // ------------------------------------------------------------

  function scanReactReferences(doc, framePath) {
    const results = [];
    const roots = [];
    let scannedElements = 0;

    let elements = [];

    try {
      elements = Array.from(
        doc.querySelectorAll(
          "canvas, [class*='chart'], [class*='position'], [class*='trade'], [class*='order']"
        )
      ).slice(0, 160);
    } catch (_) {}

    for (let i = 0; i < elements.length; i++) {
      const el = elements[i];
      scannedElements++;

      let keys = [];

      try {
        keys = Object.getOwnPropertyNames(el);
      } catch (_) {
        continue;
      }

      const reactKeys = keys.filter(key =>
        key.startsWith("__reactProps$") ||
        key.startsWith("__reactFiber$")
      );

      for (const key of reactKeys) {
        const value = getOwnValue(el, key);
        if (!value || typeof value !== "object") continue;

        if (key.startsWith("__reactProps$")) {
          roots.push({
            value,
            path: framePath + ".DOM[" + i + "].ReactProps"
          });
        }

        if (key.startsWith("__reactFiber$")) {
          for (const propName of ["memoizedProps", "pendingProps"]) {
            const props = getSafeProperty(value, propName);

            if (props && typeof props === "object") {
              roots.push({
                value: props,
                path: framePath + ".DOM[" + i + "].Fiber." + propName
              });
            }
          }
        }

        results.push({
          frame: framePath,
          elementIndex: i,
          tagName: el.tagName || "",
          className: String(el.className || "").slice(0, 160),
          reactReference: key.startsWith("__reactProps$")
            ? "React props"
            : "React fiber"
        });
      }
    }

    const functions = [];
    const seen = new WeakSet();
    let visited = 0;

    function walk(obj, path, depth) {
      if (!obj || depth > 4 || visited > 800) return;
      if (typeof obj !== "object" && typeof obj !== "function") return;
      if (seen.has(obj)) return;

      seen.add(obj);
      visited++;

      let keys = [];

      try {
        keys = Object.getOwnPropertyNames(obj).slice(0, 100);
      } catch (_) {
        return;
      }

      for (const key of keys) {
        if (SENSITIVE_KEY.test(key)) continue;

        const value = getOwnValue(obj, key);
        if (value === undefined) continue;

        const nextPath = path + "." + key;

        if (typeof value === "function") {
          if (
            /onClick|onPointer|onMouse|close|reverse|position|order|trade|submit/i.test(key)
          ) {
            const info = analyzeFunction(value, nextPath);

            if (info) {
              functions.push({
                frame: framePath,
                ...info
              });
            }
          }
        } else if (
          value &&
          typeof value === "object" &&
          !isDOMObject(value, doc.defaultView)
        ) {
          walk(value, nextPath, depth + 1);
        }
      }
    }

    for (const root of roots.slice(0, 100)) {
      walk(root.value, root.path, 0);
    }

    return {
      scannedElements,
      reactReferences: results.slice(0, 150),
      handlerFunctions: functions.slice(0, 100)
    };
  }

  // ------------------------------------------------------------
  // Search already-loaded inline JavaScript only.
  // No external files are downloaded.
  // ------------------------------------------------------------

  function scanInlineScripts(doc, framePath) {
    const matches = [];
    const externalScripts = [];

    let scripts = [];

    try {
      scripts = Array.from(doc.scripts || []);
    } catch (_) {}

    for (let i = 0; i < scripts.length; i++) {
      const script = scripts[i];

      if (script.src) {
        const src = sanitizeURL(script.src);

        if (
          /chart|trading|position|order|trade|contract|future|api|chunk/i.test(src)
        ) {
          externalScripts.push({
            index: i,
            src
          });
        }

        continue;
      }

      let source = "";

      try {
        source = script.textContent || "";
      } catch (_) {}

      if (!source || source.length > 1800000) continue;

      let count = 0;

      for (const pattern of SOURCE_PATTERNS) {
        pattern.lastIndex = 0;

        let match;

        while ((match = pattern.exec(source)) !== null) {
          const start = Math.max(0, match.index - 220);
          const end = Math.min(
            source.length,
            match.index + match[0].length + 300
          );

          matches.push({
            frame: framePath,
            scriptIndex: i,
            matchedTerm: match[0],
            snippet: cleanText(source.slice(start, end), 650)
          });

          count++;

          if (count >= 15) break;
        }

        if (count >= 15) break;
      }
    }

    return {
      totalScripts: scripts.length,
      externalScripts: externalScripts.slice(0, 100),
      inlineScriptMatches: matches.slice(0, 100)
    };
  }

  // ------------------------------------------------------------
  // Capture loaded resource names only.
  // No request is made and no response body is read.
  // ------------------------------------------------------------

  function scanResourceNames(win) {
    const results = [];

    try {
      const entries = win.performance.getEntriesByType("resource");

      for (const entry of entries) {
        if (
          /position|order|trade|contract|future|market|chart|kline|api|chunk/i.test(entry.name)
        ) {
          results.push({
            name: sanitizeURL(entry.name),
            initiatorType: entry.initiatorType || "",
            durationMs: Math.round(entry.duration || 0)
          });
        }

        if (results.length >= 100) break;
      }
    } catch (_) {}

    return results;
  }

  function runAudit() {
    const auditStarted = new Date().toISOString();
    const frameItems = collectFrames(rootWindow);

    const report = {
      reportName: "BITUNIX_CLOSE_REVERSE_API_AUDIT",
      version: VERSION,

      scanStartedAt: auditStarted,
      scanCompletedAt: null,

      },

        accessibleDocuments: 0,
        chartDocuments: 0,
        functionMatches: 0,
        reactHandlerMatches: 0,
        scriptMatches: 0

    };

    for (const item of frameItems) {
      if (!item.accessible) {
        report.documents.push(item);
        continue;
      }

      report.summary.accessibleDocuments++;

      if (item.isChart) {
        report.summary.chartDocuments++;
      }

      const objectScan = scanObjectRoots(
        item.win,
        item.doc,
        item.path
      );

      const reactScan = scanReactReferences(
        item.doc,
        item.path
      );

      const scriptScan = scanInlineScripts(
        item.doc,
        item.path
      );

      const resources = scanResourceNames(item.win);

      report.summary.functionMatches +=
        objectScan.functionMatches.length;

      report.summary.reactHandlerMatches +=
        reactScan.handlerFunctions.length;

      report.summary.scriptMatches +=
        scriptScan.inlineScriptMatches.length;

      report.documents.push({
        path: item.path,
        url: item.url,
        title: item.title,
        isChart: item.isChart,
        canvasCount: item.canvasCount,
        chartMarkers: item.chartMarkers,
        canvases: item.canvases,

        objectScan,
        reactScan,
        scriptScan,
        loadedResourceNames: resources
      });
    }

    report.scanCompletedAt = new Date().toISOString();

    return report;
  }

        Number(propertyValue(collection, "length") || 0),

        } else {

            type: typeOfValue(value)

          i++;
        }

        return result;
      }

      result.note = "Map entries could not be inspected";

          safePositionObject(value, path + "." + key)
        );

          key,
          type: typeOfValue(value)
        });

### ۱. مسیر واقعی Close

متد زیر در Broker Adapter وجود دارد:

```javascript
adapter.closePosition(positionId, quantity)
```

طبق کد استخراج‌شده:

- ورودی اول شناسه پوزیشن است.
- ورودی دوم اختیاری است و مقدار حجمی است که باید بسته شود، نه درصد.
- اگر مقدار دوم ارسال نشود، کل پوزیشن بسته می‌شود.
- بستن درصدی به فعال بودن قابلیت `supportPartialClosePosition` وابسته است.
- اگر Broker متد اختصاصی Close نداشته باشد، سیستم سفارش مخالف با ویژگی بستن پوزیشن ایجاد می‌کند.

پس از نظر ساختار نرم‌افزار، می‌توانیم در افزونه درصد دلخواه را به حجم تبدیل کنیم و مسیر بستن جزئی را استفاده کنیم؛ البته ابتدا باید فعال بودن قابلیت آن را تأیید کنیم.

### ۲. مسیر واقعی Reverse

این متد هم وجود دارد:

```javascript
adapter.reversePosition(positionId)
```

منطق آن به این شکل است:
- پوزیشن را با شناسه پیدا می‌کند.
- اگر Reverse بومی صرافی پشتیبانی شود، از آن استفاده می‌کند.
- در غیر این صورت، سفارش مخالف با حجم دو برابر حجم فعلی ایجاد می‌کند تا پوزیشن در جهت معکوس قرار بگیرد.

این دقیقاً همان مسیری است که می‌خواستیم پیدا کنیم.

        found: true,
        path,
        type: "accessor",

      path,
      ownerLevel: d.level,
      type: typeof fn,
      functionName: typeof fn === "function" ? (fn.name || "") : "",
      argumentCount: typeof fn === "function" ? fn.length : null,
      source: typeof fn === "function" ? source(fn, 8000) : null

      path,

گزارش ۱۹ یک کشف مهم دارد: **Event Bus اصلی Bitunix و Event Bus مربوط به Broker دقیقاً یک آبجکت هستند.** یعنی دو مسیر جدا نداریم و هر دو به یک سیستم فرمان‌دهی می‌رسند.

مهم‌تر اینکه مسیر درست Close را هم از مسیر اشتباه `marketClose` تفکیک کردیم:

| رویداد | کاربرد |
|---|---|
| `closePosition` | رسیدگی به بستن پوزیشن |
| `reversePosition` | اجرای Reverse پوزیشن |
| `marketClose` | تغییر وضعیت سفارش سریع بازار؛ نه فرمان بستن پوزیشن |

Listener مربوط به Close این ورودی را انتظار دارد:

```javascript
{
  position,
  resolve,
  reject
}
```

و Listener مربوط به Reverse ورودی `positionId` و `resolve` را دریافت می‌کند و سپس مسیر Reverse را فراخوانی می‌کند.

## مرحله بعد: بررسی شکل دقیق آبجکت پوزیشن

یک تفاوت مهم باقی مانده: Listener مربوط به Close از `position.id` استفاده می‌کند، اما Listener مربوط به Reverse از `position.positionId`. باید مشخص کنیم آبجکت پوزیشن فعلی هر دو فیلد را دارد یا باید برای هر عملیات آبجکت متفاوتی بسازیم.

این کد فقط همین موضوع را بررسی می‌کند و پنجره گزارش با دکمه کپی باز می‌کند.

  const D = W.document;

  const old = D.getElementById(PANEL_ID);

      networkRequests: 0,
      positionIdsExposed: false,
      exactQuantitiesExposed: false
    },
    currentPositions: [],
    eventBus: {},
    dialogWrappers: {},
    conclusion: {}
  };

  function desc(obj, key) {
    let p = obj;

    for (let level = 0; p && level < 6; level++) {
      try {
        const d = Object.getOwnPropertyDescriptor(p, key);
        if (d) return {
          d,
          level,
          value: Object.prototype.hasOwnProperty.call(d, "value")
            ? d.value : undefined
        };
        p = Object.getPrototypeOf(p);
      } catch (_) { break; }
    }

    return null;
  }

  function val(obj, key) {
    const d = desc(obj, key);
    return d && Object.prototype.hasOwnProperty.call(d.d, "value")
      ? d.value : undefined;
  }

  function keys(obj) {
    try { return Object.getOwnPropertyNames(obj); }
    catch (_) { return []; }
  }

  function fnSource(fn, max = 5000) {
    if (typeof fn !== "function") return null;
    try {
      const s = Function.prototype.toString.call(fn);
      return s.includes("[native code]") ? null : s.slice(0, max);
    } catch (_) { return null; }
  }

  function fieldInfo(obj, name) {
    const d = desc(obj, name);

    if (!d) return { present: false };

    if (!Object.prototype.hasOwnProperty.call(d.d, "value")) {
      return {
        present: true,
        type: "accessor",
        getterInvoked: false
      };
    }

    const v = d.value;

    if (name === "id" || name === "positionId") {
      return {
        present: v !== undefined && v !== null,
        type: typeof v,
        valueHidden: true,
        stringLength: typeof v === "string" ? v.length : null
      };
    }

    if (name === "qty" || name === "qtyBySide") {
      return {
        present: v !== undefined && v !== null,
        type: typeof v,
        valueHidden: true,
        sign: typeof v === "number"
          ? v > 0 ? "positive" : v < 0 ? "negative" : "zero"
          : null
      };
    }

    if (v === null || ["string", "number", "boolean"].includes(typeof v)) {
      return { present: true, type: typeof v, value: v };
    }

    return {
      present: true,
      type: Array.isArray(v) ? "array" : typeof v
    };
  }

  function nestedKeys(obj, name) {
    const v = val(obj, name);
    if (!v || typeof v !== "object") {
      return { present: false };
    }

**پیشنهاد من:** یک گزارش تشخیصی نهایی برای همین Callback بسازیم؛ بدون ارسال سفارش. بعد از آن می‌توانیم پیاده‌سازی دو دکمه `CLOSE` و `REVERSE` را شروع کنیم.
