## ۴. استخراج بعدی باید چه چیزی را مشخص کند؟

به نظرم مرحله بعد را کاملاً به شناسایی جریان معاملات اختصاص بدهیم.

این موارد را باید دقیق استخراج کنیم:

| مورد | چیزی که باید بفهمیم |
|---|---|
| REST معاملات | آدرس و پارامتر درخواست |
| Limit | حداکثر تعداد معاملات قابل دریافت |
| Trade ID | شناسه یکتای هر معامله |
| Timestamp | دقت زمانی میلی‌ثانیه‌ای |
| Price | قیمت هر معامله |
| Quantity | حجم معامله |
| Side | خرید یا فروش |
| WebSocket | آدرس و ساختار پیام‌های زنده |
| Latency | فاصله زمانی دریافت معامله تا رسیدن آن به مرورگر |

**مخصوصاً Trade ID و Timestamp برای جلوگیری از ثبت دوباره معاملات در کندل‌ها مهم هستند.**

  const usefulKey =

  function summarize(value, depth = 0) {
    if (
      value === null ||
      typeof value === 'number' ||
      typeof value === 'boolean'
    ) return value;

    if (typeof value === 'string') {
      return clean(value, 140);
    }

    if (typeof value === 'bigint') {
      return String(value);
    }

    if (value === undefined) return null;

    if (depth > 4) {
      return '[depth-limit]';
    }

    if (Array.isArray(value)) {
      return {
        _type: 'array',
        length: value.length,
        sample: value.slice(0, 6).map(v =>
          summarize(v, depth + 1)
        )
      };
    }

    if (typeof value !== 'object') {
      return `[${typeof value}]`;
    }

    const keys = safe(
      () => Object.keys(value),
      []
    ).filter(k => !sensitive.test(k));

    const result = {
      _type: 'object',
      keys: keys.slice(0, 65)
    };

    const fields = {};

    for (const key of keys.slice(0, 65)) {
      if (!usefulKey.test(key)) continue;

      const v = safe(() => value[key]);

      if (
        v === null ||
        ['string', 'number', 'boolean'].includes(typeof v)
      ) {
        fields[key] = summarize(v, depth + 1);
      } else if (
        v !== undefined &&
        (Array.isArray(v) || typeof v === 'object')
      ) {
        fields[key] = summarize(v, depth + 1);
      }
    }

    result.fields = fields;

    return result;
  }

  function findTradeCandidates(
    value,
    path = '$',
    depth = 0,
    output = []
  ) {
    if (!value || depth > 5 || output.length >= 12) {
      return output;
    }

    if (Array.isArray(value)) {
      value.slice(0, 40).forEach((item, i) => {
        findTradeCandidates(
          item,
          `${path}[${i}]`,
          depth + 1,
          output
        );
      });

      return output;
    }

    if (typeof value !== 'object') {
      return output;
    }

    const keys = Object.keys(value);

    const hasPrice = keys.some(k =>
      /^(price|p|px|lastPrice|dealPrice)$/i.test(k)
    );

    const hasTime = keys.some(k =>
      /^(timestamp|time|ts|t|T|tradeTime|createTime)$/i.test(k)
    );

    const hasQuantity = keys.some(k =>
      /^(qty|quantity|size|volume|v|amount)$/i.test(k)
    );

    if (hasPrice && hasTime && hasQuantity) {
      output.push({
        path,
        keys: keys.filter(k => !sensitive.test(k)),
        data: summarize(value)
      });
    }

    for (const key of keys.slice(0, 35)) {
      if (sensitive.test(key)) continue;

      const child = safe(() => value[key]);

      if (
        child &&
        typeof child === 'object'
      ) {
        findTradeCandidates(
          child,
          path + '.' + key,
          depth + 1,
          output
        );
      }
    }

    return output;
  }

    if (raw.length > 220000) {
      return {

        parseable: true,
        json: JSON.parse(raw)
      };
    } catch (_) {
      return {
        parseable: false,
        length: raw.length
      };

  function bodySummary(body) {
    if (body === undefined || body === null) return null;

    if (typeof body === 'string') {
      if (body.length > 50000) {
        return {
          type: 'string',
          length: body.length,
          tooLarge: true

  function addHttpRequest(info) {

      response: null

    report.httpRequests.push(entry);

  function processHttpResponse(entry, responseText, status) {
    if (!entry) return;

    report.counters.httpResponses++;

    let parsed;

    try {
      parsed = JSON.parse(responseText);
    } catch (_) {
      entry.response = {
        status,
        parseableJSON: false,
        length: responseText?.length || 0
      };
      return;
    }

    const candidates = findTradeCandidates(parsed);

    entry.response = {
      status,
      parseableJSON: true,
      structure: summarize(parsed),
      tradeCandidateCount: candidates.length,
      tradeSamples: candidates.slice(0, 5)
    };

    for (const item of candidates) {
      report.counters.tradeCandidateObjects++;

      if (report.tradeCandidates.length < MAX_TRADES) {
        report.tradeCandidates.push({
          source: 'HTTP',
          requestPath: entry.url.path,
          capturedAt: new Date().toISOString(),
          ...item
        });
      }
    }
  }

  function patchNetwork(realm, realmName) {
    if (!realm || stopped) return;

    // Fetch
    const originalFetch = safe(() => realm.fetch);

    if (typeof originalFetch === 'function') {
      const wrappedFetch = function(input, init) {
        const rawURL = safe(
          () => typeof input === 'string'
            ? input
            : input.url,
          ''
        );

        const methodName = clean(
          init?.method ||
          safe(() => input.method) ||
          'GET',
          12
        ).toUpperCase();

        let entry = null;

        if (isInteresting(rawURL, realm.location?.href)) {
          entry = addHttpRequest({
            realm: realmName,
            transport: 'fetch',
            method: methodName,
            url: safeURL(rawURL, realm.location?.href),
            requestBody: bodySummary(init?.body)
          });
        }

        const promise = originalFetch.apply(this, arguments);

        if (!entry) return promise;

        return Promise.resolve(promise).then(response => {
          try {
            const clone = response.clone();

            addPending(
              clone.text().then(text => {
                processHttpResponse(
                  entry,
                  text,
                  response.status
                );
              })
            );
          } catch (e) {
            noteError('fetch-response', e);
          }

          return response;
        });
      };

      try {
        realm.fetch = wrappedFetch;

        restorers.push(() => {
          if (realm.fetch === wrappedFetch) {
            realm.fetch = originalFetch;
          }
        });
      } catch (e) {
        noteError('patch-fetch-' + realmName, e);
      }
    }

    // XMLHttpRequest
    const XHR = safe(() => realm.XMLHttpRequest);
    const proto = safe(() => XHR.prototype);

    if (!proto) return;

    const originalOpen = proto.open;
    const originalSend = proto.send;

    if (
      typeof originalOpen === 'function' &&
      typeof originalSend === 'function'
    ) {
      const metadata = new WeakMap();

      const wrappedOpen = function(methodName, url) {
        metadata.set(this, {
          method: clean(methodName, 12).toUpperCase(),
          url: String(url || '')
        });

        return originalOpen.apply(this, arguments);
      };

      const wrappedSend = function(body) {
        const meta = metadata.get(this);
        let entry = null;

        if (
          meta &&
          isInteresting(meta.url, realm.location?.href)
        ) {
          entry = addHttpRequest({
            realm: realmName,
            transport: 'xhr',
            method: meta.method,
            url: safeURL(meta.url, realm.location?.href),
            requestBody: bodySummary(body)
          });

          this.addEventListener('load', () => {
            try {
              const type = this.responseType;
              let responseText = '';

              if (type === 'json') {
                responseText = JSON.stringify(this.response);
              } else if (type === '' || type === 'text') {
                responseText = this.responseText;
              } else {
                entry.response = {
                  status: this.status,
                  responseType: type,
                  captured: false
                };
                report.counters.httpResponses++;
                return;
              }

              processHttpResponse(
                entry,
                responseText,
                this.status
              );
            } catch (e) {
              noteError('xhr-response', e);
            }
          }, { once: true });
        }

        return originalSend.apply(this, arguments);
      };

      try {
        proto.open = wrappedOpen;
        proto.send = wrappedSend;

        restorers.push(() => {
          if (proto.open === wrappedOpen) {
            proto.open = originalOpen;
          }

          if (proto.send === wrappedSend) {
            proto.send = originalSend;
          }
        });
      } catch (e) {
        noteError('patch-xhr-' + realmName, e);
      }
    }
  }

  function isRelevantMessage(text, parsed, candidates) {
    if (
      typeof text === 'string' &&
      /XAUTUSDT/i.test(text)
    ) return true;

    if (candidates.length > 0) return true;

    if (
      typeof text === 'string' &&
      /\b(trade|deal|tick|market|ticker|agg|realtime)\b/i.test(text)
    ) return true;

    if (
      parsed &&
      parsed.parseable &&
      /trade|deal|tick|market|ticker|agg|realtime/i.test(
        JSON.stringify(parsed.json).slice(0, 5000)
      )
    ) return true;

    return false;
  }

  function captureWSMessage(event, ws, realmName) {
    if (stopped) return;

    report.counters.websocketMessages++;

    const data = event?.data;

    if (typeof data !== 'string') {
      report.counters.binaryMessages++;

      const info = {
        capturedAt: new Date().toISOString(),
        socket: safeURL(
          safe(() => ws.url, ''),
          location.href
        ),
        type: data?.constructor?.name || typeof data,
        byteLength: safe(() => data.byteLength, null)
      };

      if (report.socketMessages.length < MAX_WS) {
        report.socketMessages.push(info);
      }

      return;
    }

    const decoded = parsePayload(data);

    if (!decoded.parseable) return;

    const candidates = findTradeCandidates(decoded.json);

    if (!isRelevantMessage(data, decoded, candidates)) {
      return;
    }

    report.counters.matchingWebsocketMessages++;

    if (report.socketMessages.length < MAX_WS) {
      report.socketMessages.push({
        capturedAt: new Date().toISOString(),
        realm: realmName,
        socket: safeURL(
          safe(() => ws.url, ''),
          location.href
        ),
        messageLength: data.length,
        structure: summarize(decoded.json),
        tradeCandidateCount: candidates.length
      });
    }

    recordTradeCandidates(decoded.json, 'WebSocket');
  }

  function attachSocket(ws, realmName) {
    if (!ws || stopped || seenSockets.has(ws)) return;

    const isSocket = safe(() => {
      return (
        typeof ws.addEventListener === 'function' &&
        typeof ws.send === 'function' &&
        typeof ws.readyState === 'number' &&
        typeof ws.url === 'string'
      );
    }, false);

    if (!isSocket) return;

    seenSockets.add(ws);

    const handler = event =>
      captureWSMessage(event, ws, realmName);

    try {
      ws.addEventListener('message', handler);

      socketListeners.push({
        ws,
        handler
      });

      report.sockets.push(socketInfo(ws, realmName));

      report.counters.socketsObserved++;
    } catch (e) {
      noteError('attach-socket', e);
    }
  }

  function findChartRoots(frameWindow) {
    const roots = [];

    const get = (obj, key) => getValue(obj, key);

    const collection =
      get(frameWindow, 'chartWidgetCollection');

    const widget =
      get(frameWindow, 'chartWidget');

    const chartApi =
      get(frameWindow, 'ChartApiInstance');

    const tvApi =
      get(frameWindow, 'tradingViewApi');

    if (collection) roots.push(collection);
    if (widget) roots.push(widget);
    if (chartApi) roots.push(chartApi);
    if (tvApi) roots.push(tvApi);

    let activeWidget = null;

    if (collection) {
      activeWidget = get(collection, 'activeChartWidget');

      if (!activeWidget) {
        const getAll = collection.getAll;

        if (typeof getAll === 'function') {
          const all = safe(() => getAll.call(collection), []);
          if (Array.isArray(all)) {
            activeWidget = all[0] || null;
          }
        }
      }
    }

    if (activeWidget) roots.push(activeWidget);

    const model =
      widget?.model?.() ||
      activeWidget?.model?.();

    if (model) {
      roots.push(model);

      const series = safe(
        () => model.mainSeries(),
        null
      );

      if (series) {
        roots.push(series);

        const source =
          get(series, '_seriesSource') ||
          safe(() => series.seriesSource(), null);

        if (source) {
          roots.push(source);

          const gateway = get(source, '_gateway');

          if (gateway) {
            roots.push(gateway);

            const gatewayChartApi =
              get(gateway, '_chartApi');

            if (gatewayChartApi) {
              roots.push(gatewayChartApi);
            }
          }
        }
      }
    }

    const engine = chartApi
      ? get(chartApi, '_studyEngine')
      : null;

    if (engine) {
      roots.push(engine);

      const feed = get(engine, '_externalDatafeed');

      if (feed) roots.push(feed);
    }

    return roots.filter(Boolean);
  }

  function scanObjectGraph(roots, frameWindow, realmName) {
    const queue = roots.map(obj => ({
      obj,
      depth: 0
    }));

    const visited = new WeakSet();
    let nodeCount = 0;

    while (
      queue.length &&
      nodeCount < 1800 &&
      !stopped
    ) {
      const current = queue.shift();
      const obj = current.obj;

      if (
        !obj ||
        (typeof obj !== 'object' &&
         typeof obj !== 'function')
      ) continue;

      if (visited.has(obj)) continue;
      visited.add(obj);
      nodeCount++;

      attachSocket(obj, realmName);

      if (current.depth >= 7) continue;

      if (
        obj instanceof frameWindow.Element ||
        obj instanceof frameWindow.Window ||
        obj instanceof frameWindow.Document
      ) continue;

      if (obj instanceof frameWindow.Map) {
        let n = 0;

        for (const [k, v] of obj.entries()) {
          if (n++ >= 35) break;

          if (k && typeof k === 'object') {
            queue.push({
              obj: k,
              depth: current.depth + 1
            });
          }

          if (v && typeof v === 'object') {
            queue.push({
              obj: v,
              depth: current.depth + 1
            });
          }
        }

        continue;
      }

      const names = safe(
        () => Object.getOwnPropertyNames(obj),
        []
      );

      for (const key of names.slice(0, 100)) {
        if (sensitive.test(key)) continue;

        const d = safe(
          () => Object.getOwnPropertyDescriptor(obj, key)
        );

        if (
          !d ||
          !Object.prototype.hasOwnProperty.call(d, 'value')
        ) continue;

        const value = d.value;

        if (
          value &&
          typeof value === 'object' &&
          !visited.has(value)
        ) {
          queue.push({
            obj: value,
            depth: current.depth + 1
          });
        }
      }
    }

    return nodeCount;
  }

  // ------------------------------------------------
  // Temporarily observe future WebSocket instances
  // ------------------------------------------------

  function patchWebSocket(realm, realmName) {
    const NativeWS = safe(() => realm.WebSocket);

    if (typeof NativeWS !== 'function') return;

    const WrappedWS = new Proxy(NativeWS, {
      construct(target, args, newTarget) {
        const actualTarget =
          newTarget === WrappedWS
            ? target
            : newTarget;

        const ws = Reflect.construct(
          target,
          args,
          actualTarget
        );

        attachSocket(ws, realmName);

        return ws;
      }
    });

    try {
      realm.WebSocket = WrappedWS;

      restorers.push(() => {
        if (realm.WebSocket === WrappedWS) {
          realm.WebSocket = NativeWS;
        }
      });
    } catch (e) {
      noteError('patch-websocket-' + realmName, e);
    }
  }

  // ------------------------------------------------
  // 1. Inspect existing resources
  // ------------------------------------------------

  for (const realm of [
    {
      win: window,

    };

    return entry.requestId;
  }

  function summarizeResponse(text, status, contentType, requestId) {

    if (!parsed) {

        requestId,
        status,
        contentType: clean(contentType, 100),
        json: false,
        bodyLength: text ? text.length : 0,
        sample: clean(text, 160)
      });
      return;
    }

      requestId,
      status,
      contentType: clean(contentType, 100),
      json: true,
      structure: sanitizeValue(parsed),
      tradeCandidateCount: candidates.length,
      tradeSamples: candidates
    });
  }

  async function readLimitedResponse(response, maxBytes = 220000) {
    try {
      const clone = response.clone();
      if (!clone.body || !clone.body.getReader) {
        const text = await clone.text();
        return { text: text.slice(0, maxBytes), truncated: text.length > maxBytes };
      }
      const reader = clone.body.getReader();
      const decoder = new TextDecoder();
      let text = '';
      let bytes = 0;
      let truncated = false;
      while (true) {
        const part = await reader.read();
        if (part.done) break;
        const chunk = part.value;
        const remaining = maxBytes - bytes;
        if (chunk.byteLength > remaining) {
          text += decoder.decode(chunk.slice(0, Math.max(0, remaining)), { stream: true });
          bytes = maxBytes;
          truncated = true;
          try { await reader.cancel(); } catch (_) {}
          break;
        }
        bytes += chunk.byteLength;
        text += decoder.decode(chunk, { stream: true });
      }
      text += decoder.decode();
      return { text, truncated };
    } catch (_) {
      return { text: '', truncated: false, failed: true };
    }
  }

      structure: sanitizeValue(parsed),

  function attachSocket(ws, source) {
    if (!ws || state.observedSockets.has(ws)) return;
    state.observedSockets.add(ws);

    const rawUrl = (() => { try { return ws.url; } catch (_) { return ''; } })();
    const relevant = isRelevantSocketUrl(rawUrl);

      source,
      relevantMarketUrl: relevant,
      url: socketSafeUrl(rawUrl),
      protocol: clean(ws.protocol, 80),
      readyState: ws.readyState
    });

    const onClose = event => emit('websocket_close', { url: socketSafeUrl(rawUrl), code: event.code, wasClean: event.wasClean, source });
    const onError = () => emit('websocket_error', { url: socketSafeUrl(rawUrl), source });
    const onMessage = event => {
      if (!state.active) return;
      const data = event && event.data;
      if (typeof data !== 'string') {
        emit('websocket_binary_metadata', {
          url: socketSafeUrl(rawUrl),
          dataType: data && data.constructor ? data.constructor.name : typeof data,
          byteLength: data && typeof data.byteLength === 'number' ? data.byteLength : null,
          blobSize: data && typeof data.size === 'number' ? data.size : null,
          source
        });
        return;
      }
      const market = maybeMarketMessage(data);
      if (!market) return;
      emit('websocket_market_message', {
        url: socketSafeUrl(rawUrl),
        messageLength: data.length,
        ...market,
        source
      });
    };

    try {
      ws.addEventListener('open', onOpen);
      ws.addEventListener('close', onClose);
      ws.addEventListener('error', onError);
      ws.addEventListener('message', onMessage);
      state.socketListeners.push({ ws, onOpen, onClose, onError, onMessage });
    } catch (_) {}
  }

  if (typeof NativeWebSocket === 'function') {
    const WrappedWebSocket = new Proxy(NativeWebSocket, {
      construct(target, args, newTarget) {
        const ws = Reflect.construct(
          target,
          args,
          newTarget === WrappedWebSocket ? target : newTarget
        );
        attachSocket(ws, 'constructed-after-document-start');
        return ws;
      }
    });

    try {
      window.WebSocket = WrappedWebSocket;
      state.restorers.push(() => {
        if (window.WebSocket === WrappedWebSocket) window.WebSocket = NativeWebSocket;
      });
    } catch (_) {}
  }

  // ------------------------------------------------------
  // Message bridge: snapshots + STOP commands from UI
  // ------------------------------------------------------
  function stopCapture() {
    if (!state.active) {
      notifySnapshot();
      return;
    }
    emit('capture_stopped', { reason: 'user-request' });
    state.active = false;

    for (const item of state.socketListeners) {
      try {
        item.ws.removeEventListener('open', item.onOpen);
        item.ws.removeEventListener('close', item.onClose);
        item.ws.removeEventListener('error', item.onError);
        item.ws.removeEventListener('message', item.onMessage);
      } catch (_) {}
    }
    state.socketListeners.length = 0;

    for (const restore of state.restorers.reverse()) {
      try { restore(); } catch (_) {}
    }
    state.restorers.length = 0;
    notifySnapshot();
  }

  window.addEventListener('message', event => {
    const data = event && event.data;
    if (!data || data.bridge !== BRIDGE || data.target !== 'MAIN') return;

    if (data.action === 'SNAPSHOT') notifySnapshot();
    else if (data.action === 'STOP') stopCapture();
  });

  emit('observer_started', {
    note: 'Early page-world observer active; only public market-like HTTP endpoints and market-like WebSocket messages are summarized.'
  });
})();

  function isTrustedSource(event) {
    if (event.source === window) return true;
    try {
      return Array.from(document.querySelectorAll('iframe'))
        .some(frame => frame.contentWindow === event.source);
    } catch (_) {

  function receive(event) {
    const data = event && event.data;
    if (!data || data.bridge !== BRIDGE || !isTrustedSource(event)) return;

    if (data.kind === 'RECORD' && data.record && data.record.id) {
      state.records.set(data.record.id, data.record);
      if (data.record.frame) {
        const key = `${data.record.frame.origin}${data.record.frame.path}`;
        state.frames.set(key, data.record.frame);
      }
      if (data.record.type === 'capture_stopped') state.activeFrames.delete(data.record.frame?.path || '');
      renderStats();
    }

    if (data.kind === 'SNAPSHOT') {
      const key = `${data.frame?.origin || ''}${data.frame?.path || ''}`;
      if (key) state.frames.set(key, data.frame);
      if (data.active) state.activeFrames.add(key);
      else state.activeFrames.delete(key);
      for (const record of (Array.isArray(data.records) ? data.records : [])) {
        if (record && record.id) state.records.set(record.id, record);
      }
      renderStats();
    }
  }

$ErrorActionPreference = "Stop"

$source = ".\گزارشات بیتیونیکس9.txt"

if (!(Test-Path -LiteralPath $source)) {
    Write-Host "File not found: $source" -ForegroundColor Red
    exit
}

Write-Host "Reading original report..." -ForegroundColor Cyan

$data = Get-Content -LiteralPath $source -Raw -Encoding UTF8 |
    ConvertFrom-Json

function Select-Fields($object, $names) {
    $result = [ordered]@{}

    foreach ($name in $names) {
        $property = $object.PSObject.Properties[$name]

        if ($null -ne $property) {
            $result[$name] = $property.Value
        }
    }

    return $result
}

$meta = Select-Fields $data @(
    "diagnostic",
    "startedAt",
    "captureStartedAt",
    "finishedAt",
    "durationMs",
    "environment",
    "counters",
    "sockets",
    "existingResources",
    "errors",
    "safety"
)

$http = Select-Fields $data @(
    "diagnostic",
    "counters",
    "httpRequests"
)

$websocket = Select-Fields $data @(
    "diagnostic",
    "counters",
    "sockets",
    "socketMessages",
    "tradeCandidates"
)

$files = @(

        Name = "گزارش9-A-Metadata.json"
        Data = $meta
    },
    @{
        Name = "گزارش9-B-HTTP.json"
        Data = $http
    },
    @{
        Name = "گزارش9-C-WebSocket.json"
        Data = $websocket
    }
)

foreach ($item in $files) {
    $json = ConvertTo-Json -InputObject $item.Data -Depth 100

    Set-Content -LiteralPath $item.Name `
        -Value $json `
        -Encoding UTF8

    $size = (Get-Item $item.Name).Length

    Write-Host "$($item.Name) - $([math]::Round($size / 1KB, 1)) KB" `
        -ForegroundColor Green
}

Write-Host ""
Write-Host "All 3 files created successfully." -ForegroundColor Green
</CodeBlock>
