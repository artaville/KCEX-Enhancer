حتماً. این **کد کامل Order Panel Audit v2.0** است. فعلاً فقط برای بررسی پنل سفارش و پیدا کردن دکمه‌های **Close** و **Reverse** از آن استفاده می‌کنیم.

**روش اجرا:** اول یک پوزیشن کوچک باز کن تا بخش Positions نمایش داده شود. بعد در همان صفحه Bitunix کلید `F12` را بزن، وارد تب **Console** شو و تمام کد زیر را یک‌جا Paste و اجرا کن.

<CodeBlock language="javascript" editable>
(() => {
  "use strict";

  const VERSION = "2.0";
  const REPORT_NAME = "BITUNIX_ORDER_PANEL_AUDIT";

  // Start from the top page whenever same-origin access is available.
  let rootDoc = document;
  try {
    rootDoc = window.top.document;
  } catch (_) {}

  // Remove an older audit popup before scanning.
  try {
    rootDoc.getElementById("bitunix-order-audit-panel-v20")?.remove();
  } catch (_) {}

  const docs = [];
  const seenDocs = new Set();

  function addDocument(doc, frameLabel) {
    if (!doc || seenDocs.has(doc)) return;
    seenDocs.add(doc);

    docs.push({
      doc,
      frame: frameLabel
    });

    let frames = [];
    try {
      frames = Array.from(doc.querySelectorAll("iframe"));
    } catch (_) {}

    frames.forEach((frame, index) => {
      try {
        if (frame.contentDocument) {
          addDocument(
            frame.contentDocument,
            frameLabel + " > iframe[" + index + "]"
          );
        }
      } catch (_) {
        // Cross-origin frames are intentionally skipped.
      }
    });
  }

  addDocument(rootDoc, "top");

  function clean(value, max = 500) {
    return String(value ?? "")
      .replace(/\s+/g, " ")
      .trim()
      .slice(0, max);
  }

  function getClass(el) {
    if (!el) return "";
    if (typeof el.className === "string") {
      return clean(el.className, 500);
    }
    return clean(el.getAttribute?.("class"), 500);
  }

  function getText(el, max = 500) {
    if (!el) return "";
    return clean(el.innerText || el.textContent || "", max);
  }

  function isVisible(el, doc) {
    try {
      const win = doc.defaultView || window;
      const style = win.getComputedStyle(el);
      const rect = el.getBoundingClientRect();

      return (
        style.display !== "none" &&
        style.visibility !== "hidden" &&
        Number(style.opacity) !== 0 &&
        rect.width > 0 &&
        rect.height > 0
      );
    } catch (_) {
      return false;
    }
  }

  function getRect(el) {
    try {
      const r = el.getBoundingClientRect();
      return {
        x: Math.round(r.x),
        y: Math.round(r.y),
        width: Math.round(r.width),
        height: Math.round(r.height)
      };
    } catch (_) {
      return null;
    }
  }

  function getLabels(el) {
    try {
      const labels = el.labels
        ? Array.from(el.labels).map(x => getText(x, 150))
        : [];

      const parentLabel = el.closest("label");
      if (parentLabel) labels.push(getText(parentLabel, 150));

      return [...new Set(labels.filter(Boolean))];
    } catch (_) {
      return [];
    }
  }

  function getReactEventNames(el) {
    try {
      const key = Object.keys(el).find(k =>
        k.startsWith("__reactProps$")
      );

      if (!key) return [];

      const props = el[key] || {};
      return Object.keys(props).filter(k => /^on[A-Z]/.test(k));
    } catch (_) {
      return [];
    }
  }

  function getParentChain(el, depth = 3) {
    const result = [];
    let node = el;

    for (let i = 0; i < depth && node; i++) {
      result.push({
        tag: node.tagName?.toLowerCase() || "",
        id: clean(node.id, 120),
        class: getClass(node).slice(0, 250),
        text: getText(node, 250),
        html: clean(node.outerHTML, 700)
      });

      node = node.parentElement;
    }

    return result;
  }

  function getAttributes(el) {
    const names = [
      "type",
      "name",
      "placeholder",
      "aria-label",
      "title",
      "role",
      "data-testid",
      "data-test",
      "autocomplete"
    ];

    const result = {};
    names.forEach(name => {
      const value = el.getAttribute?.(name);
      if (value !== null && value !== "") {
        result[name] = clean(value, 200);
      }
    });

    return result;
  }

  function scanDocument(doc, frameLabel) {
    const result = {
      frame: frameLabel,
      url: "",
      title: "",
      buttonsFound: 0,
      buttons: [],
      inputsFound: 0,
      inputs: [],
      formsFound: 0,
      forms: [],
      relevantText: [],
      closeReverseCandidates: []
    };

    try {
      result.url = doc.location.href;
    } catch (_) {}

    result.title = clean(doc.title, 200);

    // Buttons and button-like elements.
    const buttonSelector = [
      "button",
      '[role="button"]',
      'input[type="button"]',
      'input[type="submit"]',
      'a[role="button"]',
      '[class*="close" i]',
      '[class*="reverse" i]',
      '[data-testid*="close" i]',
      '[data-testid*="reverse" i]'
    ].join(",");

    let buttonElements = [];
    try {
      buttonElements = Array.from(
        doc.querySelectorAll(buttonSelector)
      );
    } catch (_) {}

    result.buttonsFound = buttonElements.length;

    buttonElements.slice(0, 350).forEach((el, index) => {
      const text = getText(el, 250);
      const cls = getClass(el);
      const aria = clean(el.getAttribute("aria-label"), 200);
      const title = clean(el.getAttribute("title"), 200);

      const hint = (
        text + " " + cls + " " + aria + " " + title
      ).toLowerCase();

      const closeOrReverse =
        /close|reverse|平仓|反向|关闭|全平|市价平仓/.test(hint);

      const visible = isVisible(el, doc);

      // Keep visible controls and hidden controls that look relevant.
      if (!visible && !closeOrReverse) return;

      const item = {
        index,
        tag: el.tagName?.toLowerCase() || "",
        text,
        class: cls,
        id: clean(el.id, 150),
        attributes: getAttributes(el),
        visible,
        disabled: Boolean(el.disabled),
        rect: getRect(el),
        reactEvents: getReactEventNames(el),
        html: clean(el.outerHTML, 1000),
        parentChain: getParentChain(el, 3)
      };

      result.buttons.push(item);

      if (closeOrReverse) {
        result.closeReverseCandidates.push(item);
      }
    });

    // Inputs: never record current input values.
    let inputElements = [];
    try {
      inputElements = Array.from(
        doc.querySelectorAll(
          "input, textarea, select, [contenteditable='true']"
        )
      );
    } catch (_) {}

    result.inputsFound = inputElements.length;

    inputElements.slice(0, 250).forEach((el, index) => {
      const visible = isVisible(el, doc);
      const attrs = getAttributes(el);
      const cls = getClass(el);

      if (!visible && !attrs.placeholder && !attrs.name) return;

      result.inputs.push({
        index,
        tag: el.tagName?.toLowerCase() || "",
        type: attrs.type || "",
        name: attrs.name || "",
        placeholder: attrs.placeholder || "",
        ariaLabel: attrs["aria-label"] || "",
        title: attrs.title || "",
        labels: getLabels(el),
        class: cls,
        id: clean(el.id, 150),
        visible,
        disabled: Boolean(el.disabled),
        rect: getRect(el),
        parentChain: getParentChain(el, 3)
      });
    });

    // Forms: structure only, no values.
    let forms = [];
    try {
      forms = Array.from(doc.querySelectorAll("form"));
    } catch (_) {}

    result.formsFound = forms.length;

    forms.slice(0, 50).forEach((form, index) => {
      result.forms.push({
        index,
        id: clean(form.id, 120),
        class: getClass(form),
        action: clean(form.getAttribute("action"), 250),
        method: clean(form.getAttribute("method"), 50),
        visible: isVisible(form, doc),
        text: getText(form, 1000),
        inputCount: form.querySelectorAll(
          "input, textarea, select"
        ).length,
        buttons: Array.from(
          form.querySelectorAll("button, [role='button']")
        ).slice(0, 30).map(btn => ({
          text: getText(btn, 150),
          class: getClass(btn),
          disabled: Boolean(btn.disabled)
        }))
      });
    });

    // Text around positions, order actions, and close/reverse controls.
    try {
      const lines = (doc.body?.innerText || "")
        .split(/\n+/)
        .map(x => clean(x, 250))
        .filter(Boolean);

      const pattern =
        /position|positions|close|reverse|open long|open short|take profit|stop loss|\bTP\b|\bSL\b|平仓|持仓|反向|止盈|止损/i;

      result.relevantText = lines
        .filter(line => pattern.test(line))
        .slice(0, 120);
    } catch (_) {}

    return result;
  }

  const scannedDocuments = docs.map(entry =>
    scanDocument(entry.doc, entry.frame)
  );

  const report = {
    reportName: REPORT_NAME,
    version: VERSION,
    timestamp: new Date().toISOString(),
    page: (() => {
      try {
        return rootDoc.location.href;
      } catch (_) {
        return "";
      }
    })(),
    pageTitle: clean(rootDoc.title, 200),
    viewport: {
      width: rootDoc.defaultView?.innerWidth || 0,
      height: rootDoc.defaultView?.innerHeight || 0
    },
    safety: {
      readOnly: true,
      buttonsClicked: 0,
      formsSubmitted: 0,
      networkRequestsMade: 0,
      inputValuesRecorded: false,
      privateWebSocketDataRead: false
    },
    documentsScanned: scannedDocuments.length,
    summary: {
      totalButtons: scannedDocuments.reduce(
        (n, x) => n + x.buttonsFound, 0
      ),
      totalInputs: scannedDocuments.reduce(
        (n, x) => n + x.inputsFound, 0
      ),
      totalForms: scannedDocuments.reduce(
        (n, x) => n + x.formsFound, 0
      ),
      closeReverseCandidates: scannedDocuments.reduce(
        (n, x) => n + x.closeReverseCandidates.length, 0
      )
    },
    documents: scannedDocuments
  };

  const reportText = JSON.stringify(report, null, 2);

  console.log(
    "%cBITUNIX ORDER PANEL AUDIT v" + VERSION,
    "font-size:16px;font-weight:bold;color:#16a085"
  );
  console.log("Report:", report);
  console.log("Report length:", reportText.length, "characters");

  // Copyable report popup.
  const panel = rootDoc.createElement("div");
  panel.id = "bitunix-order-audit-panel-v20";

  panel.style.cssText = [
    "position:fixed",
    "z-index:2147483647",
    "top:20px",
    "right:20px",
    "width:min(760px,90vw)",
    "height:min(620px,85vh)",
    "background:#101820",
    "color:#eaf2f5",
    "border:1px solid #38505b",
    "border-radius:12px",
    "box-shadow:0 12px 45px rgba(0,0,0,.55)",
    "padding:14px",
    "display:flex",
    "flex-direction:column",
    "gap:10px",
    "font:13px Arial,sans-serif"
  ].join(";");

  const header = rootDoc.createElement("div");
  header.style.cssText =
    "display:flex;align-items:center;justify-content:space-between;gap:10px";

  const heading = rootDoc.createElement("strong");
  heading.textContent =
    "Bitunix Order Panel Audit v" + VERSION +
    " — " + reportText.length.toLocaleString() + " characters";

  const actions = rootDoc.createElement("div");
  actions.style.cssText = "display:flex;gap:8px";

  function makeButton(label, background) {
    const button = rootDoc.createElement("button");
    button.textContent = label;
    button.style.cssText = [
      "border:0",
      "border-radius:7px",
      "padding:8px 13px",
      "cursor:pointer",
      "font-weight:bold",
      "color:white",
      "background:" + background
    ].join(";");
    return button;
  }

  const copyButton = makeButton("Copy Report", "#087f69");
  const closeButton = makeButton("Close", "#49545c");

  actions.append(copyButton, closeButton);
  header.append(heading, actions);

  const status = rootDoc.createElement("div");
  status.textContent =
    "Read-only audit. No trading buttons were clicked and no input values were read.";
  status.style.cssText = "color:#9fb3bd;font-size:12px";

  const output = rootDoc.createElement("textarea");
  output.value = reportText;
  output.spellcheck = false;
  output.style.cssText = [
    "flex:1",
    "width:100%",
    "min-height:0",
    "resize:none",
    "box-sizing:border-box",
    "background:#071016",
    "color:#bce6dc",
    "border:1px solid #344b56",
    "border-radius:8px",
    "padding:10px",
    "font:12px Consolas,monospace",
    "line-height:1.45"
  ].join(";");

  copyButton.addEventListener("click", async () => {
    try {
      const clipboard = rootDoc.defaultView?.navigator?.clipboard;
      if (clipboard?.writeText) {
        await clipboard.writeText(reportText);
      } else {
        output.focus();
        output.select();
        rootDoc.execCommand("copy");
      }
      status.textContent =
        "Report copied. Paste it into a .txt file or send it to ChatGPT.";
    } catch (_) {
      output.focus();
      output.select();
      status.textContent =
        "Automatic copy was blocked. Press Ctrl+C to copy the selected report.";
    }
  });

  closeButton.addEventListener("click", () => panel.remove());

  panel.append(header, status, output);
  (rootDoc.body || rootDoc.documentElement).appendChild(panel);

  output.focus();
  output.select();

  console.log(
    "Audit complete. Use the Copy Report button in the popup."
  );
})();

(() => {
  "use strict";

  const KEY = "__BITUNIX_HOVER_AUDIT_V1__";
  const PANEL_ID = "bitunix-hover-audit-v1";

  // Stop an older copy of this diagnostic, if present.
  try {
    window[KEY]?.stop?.();
  } catch (_) {}

  const rootDoc = document;
  const docs = [];
  const seen = new Set();
  const captures = [];

  let panel = null;
  let statusEl = null;
  let outputEl = null;
  let lastHit = null;
  let tracking = true;

  function clean(value, max = 180) {
    return String(value ?? "")
      .replace(/\s+/g, " ")
      .trim()
      .slice(0, max);
  }

  function classText(el) {
    if (!el) return "";
    if (typeof el.className === "string") {
      return clean(el.className, 400);
    }
    return clean(el.getAttribute?.("class"), 400);
  }

  function safeText(el) {
    if (!el) return "";
    const value = clean(el.innerText || el.textContent || "", 100);
    return /^(close|reverse|market close|close position)$/i.test(value)
      ? value
      : "";
  }

  function readAttributes(el) {
    const result = {};
    const names = [
      "aria-label",
      "title",
      "data-tooltip",
      "data-name",
      "data-testid",
      "data-test",
      "role",
      "type"
    ];

    for (const name of names) {
      const value = el.getAttribute?.(name);
      if (value) result[name] = clean(value, 250);
    }
    return result;
  }

  function addDocument(doc, label) {
    if (!doc || seen.has(doc)) return;
    seen.add(doc);

    docs.push({ doc, label });

    let frames = [];
    try {
      frames = Array.from(doc.querySelectorAll("iframe"));
    } catch (_) {}

    frames.forEach((frame, index) => {
      try {
        if (frame.contentDocument?.documentElement) {
          addDocument(
            frame.contentDocument,
            label + " > iframe[" + index + "]"
          );
        }
      } catch (_) {
        // Cross-origin frames are skipped.
      }
    });
  }

  addDocument(rootDoc, "top");

  function nearestControl(el) {
    let node = el;

    for (let i = 0; i < 8 && node; i++) {
      if (node.nodeType !== 1) break;

      const tag = node.tagName?.toLowerCase() || "";
      const cls = classText(node);

      const interactive =
        /^(button|a|input|select|textarea)$/.test(tag) ||
        node.getAttribute?.("role") === "button" ||
        node.hasAttribute?.("aria-label") ||
        node.hasAttribute?.("title") ||
        node.hasAttribute?.("data-tooltip") ||
        node.hasAttribute?.("data-name") ||
        /button|btn|control|toolbar|close|reverse|position|order|action|tooltip|icon/i.test(cls);

      if (interactive) return node;
      node = node.parentElement;
    }

    return null;
  }

  function cssEscape(value, win) {
    try {
      return win.CSS.escape(value);
    } catch (_) {
      return String(value).replace(/[^a-zA-Z0-9_-]/g, "\\$&");
    }
  }

  function buildSelector(el, doc) {
    if (!el || el.nodeType !== 1) return "";

    const win = doc.defaultView || window;

    if (el.id) {
      const idSelector = "#" + cssEscape(el.id, win);
      try {
        if (doc.querySelectorAll(idSelector).length === 1) {
          return idSelector;
        }
      } catch (_) {}
    }

    const parts = [];
    let node = el;

    for (let depth = 0; depth < 7 && node; depth++) {
      let part = node.tagName.toLowerCase();

      if (node.id) {
        part += "#" + cssEscape(node.id, win);
        parts.unshift(part);
        break;
      }

      const classes = Array.from(node.classList || [])
        .filter(Boolean)
        .slice(0, 3);

      classes.forEach(c => {
        part += "." + cssEscape(c, win);
      });

      const parent = node.parentElement;
      if (parent) {
        const sameTag = Array.from(parent.children)
          .filter(x => x.tagName === node.tagName);

        if (sameTag.length > 1) {
          part += ":nth-of-type(" +
            (sameTag.indexOf(node) + 1) + ")";
        }
      }

      parts.unshift(part);

      const candidate = parts.join(" > ");

      try {
        if (doc.querySelectorAll(candidate).length === 1) {
          return candidate;
        }
      } catch (_) {}

      node = parent;
    }

    return parts.join(" > ");
  }

  function elementInfo(el, doc) {
    if (!el || el.nodeType !== 1) return null;

    const cls = classText(el);
    const html = clean(el.outerHTML, 1800);
    let rect = null;
    let style = null;

    try {
      const r = el.getBoundingClientRect();
      rect = {
        x: Math.round(r.x),
        y: Math.round(r.y),
        width: Math.round(r.width),
        height: Math.round(r.height)
      };

      const s = (doc.defaultView || window).getComputedStyle(el);
      style = {
        display: s.display,
        visibility: s.visibility,
        opacity: s.opacity,
        pointerEvents: s.pointerEvents
      };
    } catch (_) {}

    const parentChain = [];
    let parent = el;

    for (let i = 0; i < 7 && parent; i++) {
      parentChain.push({
        tag: parent.tagName?.toLowerCase() || "",
        id: clean(parent.id, 120),
        class: classText(parent),
        attributes: readAttributes(parent)
      });
      parent = parent.parentElement;
    }

    return {
      tag: el.tagName?.toLowerCase() || "",
      id: clean(el.id, 150),
      class: cls,
      text: safeText(el),
      attributes: readAttributes(el),
      selector: buildSelector(el, doc),
      selectorMatchCount: (() => {
        try {
          return doc.querySelectorAll(buildSelector(el, doc)).length;
        } catch (_) {
          return null;
        }
      })(),
      rect,
      style,
      // HTML is included only for a small element, not a large position row.
      outerHTML:
        html.length <= 1400 && clean(el.innerText || "", 100).length <= 80
          ? html
          : "[omitted: element HTML too large]",
      parentChain
    };
  }

  function onPointer(event, entry) {
    if (!tracking) return;

    let target = event.target;
    if (target?.nodeType !== 1) {
      target = target?.parentElement;
    }
    if (!target) return;

    // Ignore this diagnostic popup itself.
    if (entry.doc === rootDoc && panel?.contains(target)) {
      return;
    }

    // The parent document sees the iframe as one element.
    // The chart's own document has its own listener.
    if (
      entry.doc === rootDoc &&
      target.tagName?.toLowerCase() === "iframe"
    ) {
      return;
    }

    const control = nearestControl(target);
    if (!control) return;

    lastHit = {
      doc: entry.doc,
      frame: entry.label,
      time: new Date().toISOString(),
      x: event.clientX,
      y: event.clientY,
      rawTarget: target,
      control
    };

    if (statusEl) {
      const label =
        control.getAttribute?.("aria-label") ||
        control.getAttribute?.("title") ||
        control.getAttribute?.("data-tooltip") ||
        control.getAttribute?.("data-name") ||
        safeText(control) ||
        control.tagName.toLowerCase() + "." + classText(control);

      statusEl.textContent =
        "آخرین عنصر شناسایی‌شده: " +
        entry.label + " | " + clean(label, 100);
    }
  }

  function visibleTooltipCandidates(doc) {
    const result = [];

    let elements = [];
    try {
      elements = Array.from(
        doc.querySelectorAll(
          "[title], [aria-label], [data-tooltip], [class*='tooltip' i]"
        )
      );
    } catch (_) {}

    for (const el of elements.slice(0, 2500)) {
      try {
        const style = (doc.defaultView || window).getComputedStyle(el);
        const rect = el.getBoundingClientRect();

        if (
          style.display === "none" ||
          style.visibility === "hidden" ||
          Number(style.opacity) === 0 ||
          rect.width === 0 ||
          rect.height === 0
        ) continue;

        const attrs = readAttributes(el);
        const text = clean(el.innerText || el.textContent || "", 100);
        const combined = [
          text,
          attrs.title,
          attrs["aria-label"],
          attrs["data-tooltip"]
        ].join(" ");

        if (!/reverse|market\s+close|close position/i.test(combined)) {
          continue;
        }

        result.push({
          tag: el.tagName.toLowerCase(),
          class: classText(el),
          text: safeText(el),
          attributes: attrs,
          selector: buildSelector(el, doc),
          rect: {
            x: Math.round(rect.x),
            y: Math.round(rect.y),
            width: Math.round(rect.width),
            height: Math.round(rect.height)
          }
        });

        if (result.length >= 20) break;
      } catch (_) {}
    }

    return result;
  }

  function capture(label) {
    if (!lastHit) {
      statusEl.textContent =
        "هنوز عنصری ثبت نشده است. موس را روی کنترل چارت ببر.";
      return;
    }

    const hit = lastHit;
    const doc = hit.doc;

    let elementAtSavedPoint = null;
    try {
      elementAtSavedPoint = doc.elementFromPoint(hit.x, hit.y);
    } catch (_) {}

    const freshControl = nearestControl(elementAtSavedPoint);

    const record = {
      label,
      capturedAt: new Date().toISOString(),
      pointerEventAt: hit.time,
      frame: hit.frame,
      frameUrl: (() => {
        try {
          return doc.location.href;
        } catch (_) {
          return "";
        }
      })(),
      savedPointer: {
        x: Math.round(hit.x),
        y: Math.round(hit.y)
      },
      hoveredElement: elementInfo(hit.rawTarget, doc),
      nearestControl: elementInfo(hit.control, doc),
      elementAtSavedPointer: elementInfo(elementAtSavedPoint, doc),
      controlAtSavedPointer: elementInfo(freshControl, doc),
      visibleRelatedTooltips: visibleTooltipCandidates(doc)
    };

    captures.push(record);
    updateOutput();

    statusEl.textContent =
      label + " ثبت شد. تعداد ثبت‌ها: " + captures.length;
  }

  function reportObject() {
    return {
      reportName: "BITUNIX_HOVER_AUDIT",
      version: "1.0",
      generatedAt: new Date().toISOString(),
      page: (() => {
        try {
          return rootDoc.location.href;
        } catch (_) {
          return "";
        }
      })(),
      documentsScanned: docs.map(d => ({
        frame: d.label,
        url: (() => {
          try {
            return d.doc.location.href;
          } catch (_) {
            return "";
          }
        })()
      })),
      safety: {
        readOnly: true,
        buttonsOnExchangeClicked: 0,
        ordersSubmitted: 0,
        networkRequestsMade: 0,
        formValuesRead: false,
        privateAccountDataRead: false
      },
      captures
    };
  }

  function updateOutput() {
    if (outputEl) {
      outputEl.value = JSON.stringify(reportObject(), null, 2);
    }
  }

  // Install listeners in the main page and accessible chart frames.
  docs.forEach(entry => {
    entry.doc.addEventListener(
      "pointermove",
      event => onPointer(event, entry),
      true
    );

    entry.doc.addEventListener(
      "pointerover",
      event => onPointer(event, entry),
      true
    );
  });

  // Build the diagnostic popup.
  panel = rootDoc.createElement("div");
  panel.id = PANEL_ID;
  panel.style.cssText = [
    "position:fixed",
    "z-index:2147483647",
    "top:14px",
    "right:14px",
    "width:min(440px,90vw)",
    "background:#101820",
    "color:#e8f0f3",
    "border:1px solid #405965",
    "border-radius:11px",
    "box-shadow:0 8px 35px rgba(0,0,0,.5)",
    "padding:12px",
    "font:12px Arial,sans-serif",
    "display:flex",
    "flex-direction:column",
    "gap:8px"
  ].join(";");

  const heading = rootDoc.createElement("strong");
  heading.textContent = "Bitunix Hover Audit v1.0";

  statusEl = rootDoc.createElement("div");
  statusEl.textContent =
    "موس را روی کنترل چارت ببر؛ سپس دکمه ثبت مربوطه را بزن.";
  statusEl.style.cssText =
    "color:#b8ccd4;line-height:1.5;overflow-wrap:anywhere";

  const row = rootDoc.createElement("div");
  row.style.cssText =
    "display:flex;flex-wrap:wrap;gap:6px";

  function makeButton(label, background) {
    const button = rootDoc.createElement("button");
    button.type = "button";
    button.textContent = label;
    button.style.cssText = [
      "border:0",
      "border-radius:6px",
      "padding:8px 10px",
      "cursor:pointer",
      "color:white",
      "font-weight:bold",
      "background:" + background
    ].join(";");
    return button;
  }

  const closeCapture = makeButton("Capture Close", "#087f69");
  const reverseCapture = makeButton("Capture Reverse", "#2767a8");
  const copyButton = makeButton("Copy Report", "#7554a5");
  const clearButton = makeButton("Clear", "#59636b");
  const stopButton = makeButton("Stop Tracking", "#a34b43");

  closeCapture.addEventListener("click", () => capture("Close"));
  reverseCapture.addEventListener("click", () => capture("Reverse"));

  copyButton.addEventListener("click", async () => {
    updateOutput();

    try {
      await rootDoc.defaultView.navigator.clipboard.writeText(
        outputEl.value
      );
      statusEl.textContent =
        "گزارش کپی شد. آن را برای ChatGPT بفرست.";
    } catch (_) {
      outputEl.focus();
      outputEl.select();
      try {
        rootDoc.execCommand("copy");
      } catch (_) {}
      statusEl.textContent =
        "گزارش انتخاب شد؛ Ctrl+C را بزن و آن را کپی کن.";
    }
  });

  clearButton.addEventListener("click", () => {
    captures.length = 0;
    updateOutput();
    statusEl.textContent = "ثبت‌ها پاک شدند.";
  });

  stopButton.addEventListener("click", () => {
    tracking = false;
    statusEl.textContent =
      "ردیابی متوقف شد. گزارش‌های ثبت‌شده همچنان قابل کپی هستند.";
  });

  row.append(
    closeCapture,
    reverseCapture,
    copyButton,
    clearButton,
    stopButton
  );

  outputEl = rootDoc.createElement("textarea");
  outputEl.readOnly = true;
  outputEl.value = JSON.stringify(reportObject(), null, 2);
  outputEl.style.cssText = [
    "width:100%",
    "height:115px",
    "box-sizing:border-box",
    "resize:vertical",
    "background:#071016",
    "color:#bce6dc",
    "border:1px solid #344b56",
    "border-radius:7px",
    "padding:8px",
    "font:11px Consolas,monospace"
  ].join(";");

  const note = rootDoc.createElement("div");
  note.textContent =
    "این ابزار فقط عناصر زیر موس را می‌خواند؛ هیچ دکمه معاملاتی را فشار نمی‌دهد.";
  note.style.cssText = "color:#91a6af;font-size:11px";

  const closePanel = makeButton("Close Panel", "#39464f");
  closePanel.addEventListener("click", () => {
    window[KEY]?.stop?.();
  });

  panel.append(
    heading,
    statusEl,
    row,
    outputEl,
    note,
    closePanel
  );

  (rootDoc.body || rootDoc.documentElement).appendChild(panel);

  function stop() {
    tracking = false;

    docs.forEach(entry => {
      // Listener functions are removed by closing the page or re-running
      // this script; the panel can also be removed immediately.
      // No trading action or page navigation is performed.
    });

    try {
      panel?.remove();
    } catch (_) {}

    try {
      delete window[KEY];
    } catch (_) {}
  }

  window[KEY] = { stop };

  console.log(
    "Bitunix Hover Audit v1.0 started.",
    "Accessible documents:", docs.length
  );

        const value = el.getAttribute?.(name);
        if (value) result[name] = clean(value, 250);
      } catch (_) {}
    });

    return result;
  }

  function getRect(el) {
    try {
      const r = el.getBoundingClientRect();

      return {
        x: Math.round(r.x),
        y: Math.round(r.y),
        width: Math.round(r.width),
        height: Math.round(r.height)
      };
    } catch (_) {
      return null;
    }
  }

  function getFrameRect(frame) {
    if (!frame) return null;

    try {
      const r = frame.getBoundingClientRect();

      return {
        x: Math.round(r.x),
        y: Math.round(r.y),
        width: Math.round(r.width),
        height: Math.round(r.height)
      };
    } catch (_) {
      return null;
    }
  }

  function cssEscape(value, win) {
    try {
      return win.CSS.escape(value);
    } catch (_) {
      return String(value).replace(/[^a-zA-Z0-9_-]/g, "\\$&");
    }
  }

  function buildSelector(el, doc) {
    if (!el || el.nodeType !== 1) return "";

    const win = doc.defaultView || rootWin;

    if (el.id) {
      const idSelector = "#" + cssEscape(el.id, win);

      try {
        if (doc.querySelectorAll(idSelector).length === 1) {
          return idSelector;
        }
      } catch (_) {}
    }

    const parts = [];
    let node = el;

    for (let depth = 0; depth < 9 && node; depth++) {
      let part = node.tagName.toLowerCase();

      if (node.id) {
        part += "#" + cssEscape(node.id, win);
        parts.unshift(part);
        break;
      }

      const classes = Array.from(node.classList || [])
        .filter(Boolean)
        .slice(0, 4);

      classes.forEach(cls => {
        part += "." + cssEscape(cls, win);
      });

      const parent = node.parentElement;

      if (parent) {
        const sameTag = Array.from(parent.children)
          .filter(child => child.tagName === node.tagName);

        if (sameTag.length > 1) {
          part += ":nth-of-type(" +
            (sameTag.indexOf(node) + 1) + ")";
        }
      }

      parts.unshift(part);

      const candidate = parts.join(" > ");

      try {
        if (doc.querySelectorAll(candidate).length === 1) {
          return candidate;
        }
      } catch (_) {}

      node = parent;
    }

    return parts.join(" > ");
  }

  function isDiagnosticPanel(target) {
    try {
      return Boolean(panel && panel.contains(target));
    } catch (_) {
      return false;
    }
  }

  function addDocument(doc, label, frameElement = null) {
    if (!doc || seenDocs.has(doc)) return;

    seenDocs.add(doc);

    const entry = {
      doc,
      label,
      frameElement
    };

    docs.push(entry);
    attachPointerListeners(entry);

    let frames = [];

    try {
      frames = Array.from(doc.querySelectorAll("iframe"));
    } catch (_) {}

    frames.forEach((frame, index) => {
      try {
        if (frame.contentDocument?.documentElement) {
          addDocument(
            frame.contentDocument,
            label + " > iframe[" + index + "]",
            frame
          );
        }
      } catch (_) {
        // Cross-origin frames are intentionally skipped.
      }
    });
  }

  function attachPointerListeners(entry) {
    if (attachedDocs.has(entry.doc)) return;

    attachedDocs.add(entry.doc);

    function handler(event) {
      if (!tracking) return;

      const target = event.target;

      if (!target) return;

      // Ignore interactions with our own popup.
      if (
        entry.doc === rootDoc &&
        isDiagnosticPanel(target)
      ) {
        return;
      }

      // The parent document should not replace chart-frame data
      // with pointer events on the iframe element itself.
      if (
        entry.doc === rootDoc &&
        target.tagName?.toLowerCase() === "iframe"
      ) {
        return;
      }

      lastHits.set(entry.label, {
        entry,
        time: Date.now(),
        eventType: event.type,
        clientX: event.clientX,
        clientY: event.clientY,
        pointerType: event.pointerType || "unknown"
      });

      if (statusEl && entry.frameElement) {
        statusEl.textContent =
          "موس داخل چارت ثبت شد: " +
          entry.label +
          " | مختصات " +
          Math.round(event.clientX) +
          ", " +
          Math.round(event.clientY);
      }
    }

    entry.doc.addEventListener("pointermove", handler, true);
    entry.doc.addEventListener("pointerover", handler, true);

    listeners.push({
      doc: entry.doc,
      handler
    });
  }

  function elementInfo(el, doc) {
    if (!el || el.nodeType !== 1) return null;

    const win = doc.defaultView || rootWin;

    let computed = {};

    try {
      const s = win.getComputedStyle(el);

      computed = {
        display: s.display,
        visibility: s.visibility,
        opacity: s.opacity,
        pointerEvents: s.pointerEvents,
        cursor: s.cursor,
        position: s.position,
        zIndex: s.zIndex
      };
    } catch (_) {}

    const selector = buildSelector(el, doc);

    let selectorMatchCount = null;

    try {
      selectorMatchCount = doc.querySelectorAll(selector).length;
    } catch (_) {}

    let html = "";

    try {
      html = String(el.outerHTML || "").slice(0, 1100);
    } catch (_) {}

    return {
      tag: el.tagName?.toLowerCase() || "",
      id: clean(el.id, 150),
      class: classText(el),
      text: safeText(el),
      attributes: readAttributes(el),
      selector,
      selectorMatchCount,
      rect: getRect(el),
      computedStyle: computed,
      outerHTML: html
    };
  }

  function getElementStack(doc, x, y) {
    let elements = [];

    try {
      if (typeof doc.elementsFromPoint === "function") {
        elements = doc.elementsFromPoint(x, y);
      } else {
        const el = doc.elementFromPoint(x, y);
        elements = el ? [el] : [];
      }
    } catch (_) {}

    const unique = [];
    const seen = new Set();

    for (const el of elements) {
      if (!el || seen.has(el)) continue;
      seen.add(el);

      unique.push(elementInfo(el, doc));

      if (unique.length >= 12) break;
    }

    return unique;
  }

  function getAncestorChain(doc, x, y) {
    let node = null;

    try {
      node = doc.elementFromPoint(x, y);
    } catch (_) {}

    const chain = [];
    let depth = 0;

    while (node && node.nodeType === 1 && depth < 15) {
      const win = doc.defaultView || rootWin;

      let cursor = "";

      try {
        cursor = win.getComputedStyle(node).cursor;
      } catch (_) {}

      const tag = node.tagName?.toLowerCase() || "";
      const attrs = readAttributes(node);

      const buttonLike =
        /^(button|a|input|select)$/.test(tag) ||
        attrs.role === "button" ||
        attrs["aria-label"] ||
        attrs.title ||
        attrs["data-tooltip"] ||
        attrs["data-name"] ||
        cursor === "pointer" ||
        /button|btn|control|action|close|reverse|position|order/i.test(
          classText(node)
        );

      chain.push({
        depth,
        tag,
        id: clean(node.id, 120),
        class: classText(node),
        text: safeText(node, 100),
        attributes: attrs,
        cursor,
        buttonLike,
        selector: buildSelector(node, doc),
        rect: getRect(node),
        outerHTML: String(node.outerHTML || "").slice(0, 900)
      });

      node = node.parentElement;
      depth++;
    }

    return chain;
  }

  function scanRelatedTooltips(doc) {
    const result = [];
    let elements = [];

    try {
      elements = Array.from(
        doc.querySelectorAll(
          "[title], [aria-label], [data-tooltip], " +
          "[role='tooltip'], [class*='tooltip']"
        )
      );
    } catch (_) {}

    for (const el of elements.slice(0, 3000)) {
      try {
        const win = doc.defaultView || rootWin;
        const style = win.getComputedStyle(el);
        const rect = el.getBoundingClientRect();

        if (
          style.display === "none" ||
          style.visibility === "hidden" ||
          Number(style.opacity) === 0 ||
          rect.width === 0 ||
          rect.height === 0
        ) {
          continue;
        }

        const attrs = readAttributes(el);
        const text = safeText(el, 160);

        const allText = [
          text,
          attrs.title,
          attrs["aria-label"],
          attrs["data-tooltip"]
        ].join(" ");

        if (!/market\s*close|reverse/i.test(allText)) {
          continue;
        }

        result.push({
          tag: el.tagName.toLowerCase(),
          class: classText(el),
          text,
          attributes: attrs,
          selector: buildSelector(el, doc),
          rect: getRect(el),
          outerHTML: String(el.outerHTML || "").slice(0, 800)
        });

        if (result.length >= 25) break;
      } catch (_) {}
    }

    return result;
  }

  function selectChartHit() {
    const now = Date.now();

    const candidates = [];

    for (const hit of lastHits.values()) {
      if (!hit.entry.frameElement) continue;

      const age = now - hit.time;

      if (age > 120000) continue;

      candidates.push(hit);
    }

    if (!candidates.length) return null;

    // Prefer the known TradingView chart iframe (iframe[0]).
    candidates.sort((a, b) => {
      const aPreferred =
        a.entry.label.includes("iframe[0]") ? 1 : 0;

      const bPreferred =
        b.entry.label.includes("iframe[0]") ? 1 : 0;

      if (aPreferred !== bPreferred) {
        return bPreferred - aPreferred;
      }

      return b.time - a.time;
    });

    return candidates[0];
  }

  function getSelector(el, doc) {
    if (!el || el.nodeType !== 1) return "";

    const win = doc.defaultView || W;

    if (el.id) {
      const id = "#" + win.CSS.escape(el.id);

      try {
        if (doc.querySelectorAll(id).length === 1) {
          return id;
        }
      } catch (_) {}
    }

    const parts = [];
    let node = el;

    for (let i = 0; i < 9 && node; i++) {
      let part = node.tagName.toLowerCase();

      if (node.id) {
        part += "#" + win.CSS.escape(node.id);
        parts.unshift(part);
        break;
      }

      Array.from(node.classList || [])
        .filter(Boolean)
        .slice(0, 4)
        .forEach(c => {
          part += "." + win.CSS.escape(c);
        });

      const parent = node.parentElement;

      if (parent) {
        const siblings = Array.from(parent.children)
          .filter(x => x.tagName === node.tagName);

        if (siblings.length > 1) {
          part += ":nth-of-type(" +
            (siblings.indexOf(node) + 1) + ")";
        }
      }

      parts.unshift(part);

      const candidate = parts.join(" > ");

      try {
        if (doc.querySelectorAll(candidate).length === 1) {
          return candidate;
        }
      } catch (_) {}

      node = parent;
    }

    return parts.join(" > ");
  }

  function clean(v, max = 250) {
    return String(v ?? "").replace(/\s+/g, " ").trim().slice(0, max);
  }

  function getClass(el) {
    if (!el) return "";
    return typeof el.className === "string"
      ? clean(el.className, 500)
      : clean(el.getAttribute?.("class"), 500);
  }

  function getAttrs(el) {
    const out = {};
    [
      "aria-label", "aria-describedby", "title",
      "data-tooltip", "data-name", "data-testid",
      "data-test", "role", "type"
    ].forEach(name => {
      try {
        const v = el.getAttribute?.(name);
        if (v) out[name] = clean(v, 300);
      } catch (_) {}
    });
    return out;
  }

  function rectOf(el) {
    try {
      const r = el.getBoundingClientRect();
      return {
        x: Math.round(r.x),
        y: Math.round(r.y),
        width: Math.round(r.width),
        height: Math.round(r.height)
      };
    } catch (_) {
      return null;
    }
  }

  function frameRect(frame) {
    return frame ? rectOf(frame) : null;
  }

  function getSelector(el, doc) {
    if (!el || el.nodeType !== 1) return "";

    const win = doc.defaultView || W;
    const esc = s => win.CSS?.escape
      ? win.CSS.escape(s)
      : String(s).replace(/[^a-zA-Z0-9_-]/g, "\\$&");

    if (el.id) {
      const id = "#" + esc(el.id);
      try {
        if (doc.querySelectorAll(id).length === 1) return id;
      } catch (_) {}
    }

    const parts = [];
    let node = el;

    for (let i = 0; i < 9 && node; i++) {
      let part = node.tagName.toLowerCase();

      if (node.id) {
        part += "#" + esc(node.id);
        parts.unshift(part);
        break;
      }

      Array.from(node.classList || [])
        .filter(Boolean)
        .slice(0, 4)
        .forEach(c => part += "." + esc(c));

      const parent = node.parentElement;
      if (parent) {
        const siblings = Array.from(parent.children)
          .filter(x => x.tagName === node.tagName);

        if (siblings.length > 1) {
          part += ":nth-of-type(" +
            (siblings.indexOf(node) + 1) + ")";
        }
      }

      parts.unshift(part);
      const candidate = parts.join(" > ");

      try {
        if (doc.querySelectorAll(candidate).length === 1) {
          return candidate;
        }
      } catch (_) {}

      node = parent;
    }

    return parts.join(" > ");
  }

  function describe(el, doc) {
    if (!el || el.nodeType !== 1) return null;

    const win = doc.defaultView || W;
    let style = {};

    try {
      const s = win.getComputedStyle(el);
      style = {
        display: s.display,
        visibility: s.visibility,
        opacity: s.opacity,
        pointerEvents: s.pointerEvents,
        cursor: s.cursor,
        position: s.position,
        zIndex: s.zIndex
      };
    } catch (_) {}

    const sel = getSelector(el, doc);
    let matchCount = null;

    try {
      matchCount = doc.querySelectorAll(sel).length;
    } catch (_) {}

    return {
      tag: el.tagName?.toLowerCase() || "",
      id: clean(el.id, 150),
      class: getClass(el),
      text: clean(el.innerText || el.textContent || "", 140),
      attributes: getAttrs(el),
      selector: sel,
      selectorMatchCount: matchCount,
      rect: rectOf(el),
      computedStyle: style,
      outerHTML: String(el.outerHTML || "").slice(0, 1300)
    };
  }

  function getStack(doc, x, y) {
    let elements = [];
    try {
      elements = doc.elementsFromPoint(x, y);
    } catch (_) {
      try {
        const el = doc.elementFromPoint(x, y);
        if (el) elements = [el];
      } catch (_) {}
    }

    return elements.slice(0, 10).map(el => describe(el, doc));
  }

  function getAncestors(doc, x, y) {
    let node = null;
    try {
      node = doc.elementFromPoint(x, y);
    } catch (_) {}

    const result = [];
    let depth = 0;

    while (node && node.nodeType === 1 && depth < 14) {
      const win = doc.defaultView || W;
      let cursor = "";

      try {
        cursor = win.getComputedStyle(node).cursor;
      } catch (_) {}

      const attributes = getAttrs(node);
      const tag = node.tagName.toLowerCase();
      const classes = getClass(node);

      result.push({
        depth,
        tag,
        id: clean(node.id, 120),
        class: classes,
        text: clean(node.innerText || node.textContent || "", 120),
        attributes,
        cursor,
        buttonLike:
          /^(button|a|input|select)$/.test(tag) ||
          attributes.role === "button" ||
          Boolean(attributes["aria-label"]) ||
          Boolean(attributes.title) ||
          Boolean(attributes["data-tooltip"]) ||
          cursor === "pointer" ||
          /button|btn|control|action|close|reverse|position|order|icon/i.test(classes),
        selector: getSelector(node, doc),
        rect: rectOf(node),
        outerHTML: String(node.outerHTML || "").slice(0, 900)
      });

      node = node.parentElement;
      depth++;
    }

    return result;
  }

  function scanTooltips(doc) {
    let elements = [];
    try {
      elements = Array.from(doc.querySelectorAll(
        "[title], [aria-label], [data-tooltip], " +
        "[role='tooltip'], [class*='tooltip']"
      ));
    } catch (_) {}

    const result = [];

    for (const el of elements.slice(0, 2500)) {
      try {
        const win = doc.defaultView || W;
        const s = win.getComputedStyle(el);
        const r = el.getBoundingClientRect();

        if (
          s.display === "none" ||
          s.visibility === "hidden" ||
          Number(s.opacity) === 0 ||
          !r.width || !r.height
        ) continue;

        const attributes = getAttrs(el);
        const text = clean(el.innerText || el.textContent || "", 150);
        const combined = [
          text,
          attributes.title,
          attributes["aria-label"],
          attributes["data-tooltip"]
        ].join(" ");

        if (!/market\s*close|reverse/i.test(combined)) continue;

        result.push({
          tag: el.tagName.toLowerCase(),
          class: getClass(el),
          text,
          attributes,
          selector: getSelector(el, doc),
          rect: rectOf(el),
          outerHTML: String(el.outerHTML || "").slice(0, 800)
        });

        if (result.length >= 20) break;
      } catch (_) {}
    }

    return result;
  }

  function attachDoc(doc, label, frame) {
    let entry = docMap.get(doc);

    if (!entry) {

        doc,
        label,
        frame,
        isChart: false

      docMap.set(doc, entry);

      function handler(event) {
        if (!running) return;

        const target = event.target;
        if (!target) return;

        if (entry.label === "top" && panel?.contains(target)) {
          return;
        }

        if (
          entry.label === "top" &&
          target.tagName?.toLowerCase() === "iframe"

          entry,
          time: Date.now(),
          x: event.clientX,
          y: event.clientY,
          eventType: event.type
        });

        doc.addEventListener(type, handler, true);
        listeners.push({ doc, type, handler });
      });

    // was already registered.
    let frames = [];
    try {
      frames = Array.from(doc.querySelectorAll("iframe"));
    } catch (_) {}

    frames.forEach((child, index) => {
      try {
        if (child.contentDocument?.documentElement) {
          attachDoc(
            child.contentDocument,
            label + " > iframe[" + index + "]",
            child
          );
        }
      } catch (_) {}
    });
  }

  function sampleAll(elapsed) {

      let el = null;
      try {

      } catch (_) {}

      if (!traces.has(entry.label)) {
        traces.set(entry.label, []);

      list.push({
        elapsedMs: Math.round(elapsed),
        eventAgeMs: Date.now() - hit.time,
        x: Math.round(hit.x),
        y: Math.round(hit.y),
        frame: entry.label,
        element: el ? {

          class: getClass(el),
          text: clean(el.innerText || el.textContent || "", 80),
          attributes: getAttrs(el)
        } : null
      });

      if (list.length > 40) list.shift();
    }
  }

      const hit = lastHits.get(entry.label);

      if (!hit) {
        result.push({
          frame: entry.label,
          isChart: entry.isChart,
          pointerCaptured: false,
          frameRectangle: frameRect(entry.frame)

        continue;
      }

      try {

        isChart: entry.isChart,

          catch (_) { return ""; }
