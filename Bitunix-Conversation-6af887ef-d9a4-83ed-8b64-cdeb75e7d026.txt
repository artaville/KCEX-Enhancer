- هیچ سفارشی نمی‌فرستد و هیچ تنظیمی را تغییر نمی‌دهد.

  'use strict';

  const TAG = '__BITUNIX_DIAG_01__';
  const PANEL_ID = '__bitunix_diag_01_panel__';

  const previous = document.getElementById(PANEL_ID);
  if (previous) previous.remove();

  const safe = (fn, fallback = null) => {
    try {
      const value = fn();
      return value === undefined ? fallback : value;
    } catch (_) {
      return fallback;
    }
  };

  const clean = (value, max = 180) =>
    String(value || '').replace(/\s+/g, ' ').trim().slice(0, max);

  function safeURL(raw) {
    try {
      const u = new URL(raw, location.href);
      return u.origin + u.pathname;
    } catch (_) {
      return '[unavailable]';
    }
  }

  function visible(el) {
    return safe(() => {
      const r = el.getBoundingClientRect();
      const s = getComputedStyle(el);
      return r.width > 3 && r.height > 3 &&
        s.display !== 'none' &&
        s.visibility !== 'hidden';
    }, false);
  }

  function rect(el) {
    const r = safe(() => el.getBoundingClientRect());
    if (!r) return null;

    return {
      x: Math.round(r.x),
      y: Math.round(r.y),
      width: Math.round(r.width),
      height: Math.round(r.height)
    };
  }

  function describe(el) {
    return {
      tag: el.tagName,
      id: el.id || null,
      className: clean(
        typeof el.className === 'string'
          ? el.className
          : ''
      ),
      role: el.getAttribute('role'),
      ariaLabel: clean(el.getAttribute('aria-label')),
      title: clean(el.getAttribute('title')),
      testId: el.getAttribute('data-testid'),
      visible: visible(el),
      rect: rect(el),
      canvas: el.tagName === 'CANVAS'
        ? {
            width: el.width,
            height: el.height
          }
        : undefined
    };
  }

  function describeParents(el) {
    const result = [];
    let current = el;

    for (let i = 0; current && i < 6; i++) {
      result.push({
        tag: current.tagName,
        id: current.id || null,
        className: clean(
          typeof current.className === 'string'
            ? current.className
            : ''
        )
      });
      current = current.parentElement;
    }

    return result;

    errors: [],

      readOnly: true,
      noOrdersSent: true,
      noDOMInteractions: true,
      noStorageRead: true,
      noCookiesRead: true,
      noFetchOrWebSocketInterception: true,
      sensitiveInputValuesExcluded: true,
      queryStringsRemovedFromURLs: true
    }
  };

  // 1. Page information
  report.page = {
    host: location.host,
    origin: location.origin,
    path: location.pathname,
    title: document.title,
    readyState: document.readyState,
    language: navigator.language,
    online: navigator.onLine,
    userAgent: navigator.userAgent,
    viewport: {
      width: innerWidth,
      height: innerHeight,
      devicePixelRatio
    }
  };

  // 2. Find chart-related DOM elements
  const selectors = [
    'canvas',
    'iframe',
    '[id*="chart" i]',
    '[class*="chart" i]',
    '[id*="kline" i]',
    '[class*="kline" i]',
    '[id*="tradingview" i]',
    '[class*="tradingview" i]'
  ];

  const found = new Set();

  for (const selector of selectors) {
    try {
      document.querySelectorAll(selector).forEach(el => {
        if (found.size < 180) found.add(el);
      });
    } catch (_) {}
  }

  const candidates = Array.from(found)
    .filter(visible)
    .slice(0, 140);

  report.chartCandidates = candidates.map(describe);

  report.chartParents = candidates
    .slice(0, 14)
    .map(el => ({
      element: describe(el),
      parents: describeParents(el)
    }));

  // 3. Canvases
  report.canvases = Array.from(
    document.querySelectorAll('canvas')
  ).slice(0, 60).map(describe);

  // 4. Iframes
  report.iframes = Array.from(
    document.querySelectorAll('iframe')
  ).slice(0, 30).map(frame => ({
    ...describe(frame),
    src: safeURL(frame.getAttribute('src') || ''),
    sameOrigin: safe(() => {
      void frame.contentWindow.document.location.href;
      return true;
    }, false)
  }));

  // 5. Visible buttons
  report.buttons = Array.from(

    'TradingViewApi',

    'tvChart',
    '_exposed_chartWidgetCollection',
    '__INITIAL_STATE__',
    '__NEXT_DATA__'
  ];

    ...knownGlobals
  ]);

  report.chartGlobals = Array.from(inspectGlobals)
    .slice(0, 150)
    .map(name => {
      const value = safe(() => window[name]);

      return {
        name,
        type: typeof value,
        constructor: safe(
          () => value?.constructor?.name,

          : null

  const safe = (fn, fallback = null) => {
    try {
      const v = fn();

    } catch (_) {
      return fallback;
    }
  };

  const clean = (v, max = 160) =>
    String(v ?? '').replace(/\s+/g, ' ').trim().slice(0, max);

  const chartPattern =
    /chart|tradingview|trading-view|kline|candle|lightweight|tv-widget|series|pane|plot|pricechart|datafeed/i;

    /chart|widget|study|series|symbol|resolution|interval|time|pane|price|tv|trading|datafeed|feed|kline|candle|model|source|scale|renderer/i;

  function typeOf(v) {
    if (v === null) return 'null';
    if (Array.isArray(v)) return 'Array';
    if (typeof v !== 'object' && typeof v !== 'function') {
      return typeof v;
    }

    return safe(() => {
      const p = Object.getPrototypeOf(v);
      const d = p && Object.getOwnPropertyDescriptor(p, 'constructor');
      return d && typeof d.value === 'function'
        ? d.value.name || 'Object'
        : 'Object';
    }, 'Object');
  }

    return safe(() => Object.getOwnPropertyNames(obj), []);

    const d = safe(() => Object.getOwnPropertyDescriptor(obj, key));

      return undefined;
    }

    return d.value;

  function inspectObject(obj, path, depth = 0, seen = new WeakSet()) {
    if (obj === null || obj === undefined) {
      return { path, type: String(obj) };
    }

    if (
      (typeof obj !== 'object' && typeof obj !== 'function') ||
      depth > 3

        path,
        type: typeof obj,
        value: typeof obj === 'string'

          : undefined
      };

      return { path, circular: true };

      path,
      type: typeOf(obj),

      relevantChildren: []

            path + '.' + key,
            depth + 1,

      }

    }

      panes: []
    },
    scripts: [],
    chartResources: [],
    result: {},

      readOnly: true,
      noOrdersSent: true,
      noClickActions: true,
      noPageFetch: true,
      noWebSocketCreated: true,
      noStorageRead: true,
      noCookiesRead: true,
      noInputValuesRead: true,
      noChartMutation: true,
      sensitivePropertyNamesExcluded: true
    }
  };

  function propertyNames(obj) {
    return safe(
      () => Object.getOwnPropertyNames(obj),
      []
    );
  }

  function descriptor(obj, key) {
    return safe(
      () => Object.getOwnPropertyDescriptor(obj, key)
    );
  }

  function rect(el) {
    return safe(() => {
      const r = el.getBoundingClientRect();

      return {
        x: Math.round(r.x),
        y: Math.round(r.y),
        width: Math.round(r.width),
        height: Math.round(r.height)
      };
    });
  }

  function classes(el) {
    return clean(
      typeof el.className === 'string'
        ? el.className
        : el.getAttribute('class') || '',
      200
    );
  }

  function info(el) {
    const a = {};

    for (const attr of safe(
      () => Array.from(el.attributes),
      []
    ).slice(0, 18)) {
      if (sensitive.test(attr.name)) continue;
      a[attr.name] = clean(attr.value, 130);

    return {
      tag: el.tagName,
      id: el.id || null,
      className: classes(el),
      text: clean(el.innerText || el.textContent, 100),
      aria: clean(el.getAttribute('aria-label')),
      title: clean(el.getAttribute('title')),
      rect: rect(el),
      attributes: a
    };

      report.chartWindow = {

      const knownNames = [
        'TradingView',
        'TradingViewApi',

        'tvChart',
        'widget',
        'chart',
        'Datafeeds',
        'datafeed',
        'ChartingLibraryWidget',

      const selected = new Set([
        ...knownNames,
        ...names.filter(k => relevant.test(k) && !sensitive.test(k))

      // --------------------------------------------
      // 3. Inspect object properties and methods
      // --------------------------------------------

      function inspectObject(obj, path) {
        if (
          !obj ||
          (typeof obj !== 'object' &&
           typeof obj !== 'function')
        ) {
          return {
            path,
            type: typeof obj
          };
        }

        const own = propertyNames(obj)
          .filter(k => !sensitive.test(k))
          .slice(0, 180);

        const properties = [];
        const functions = [];
        const childObjects = [];

        for (const key of own) {
          const d = descriptor(obj, key);

          if (!d) continue;

          const value = Object.prototype.hasOwnProperty.call(d, 'value')
            ? d.value
            : undefined;

          const type = d.get || d.set
            ? 'accessor'
            : typeof value;

          const entry = {
            name: key,
            type,
            constructor: value !== undefined
              ? safe(() => value.constructor?.name)
              : null
          };

          if (typeof value === 'function') {
            entry.functionName = clean(value.name, 80);
            functions.push(entry);
          }

          if (
            value !== null &&
            value !== undefined &&
            (typeof value === 'object' ||
             typeof value === 'function') &&
            relevant.test(key)
          ) {
            childObjects.push({
              name: key,
              type,
              constructor: entry.constructor,
              keys: propertyNames(value)
                .filter(k => !sensitive.test(k))
                .slice(0, 60)
            });
          }

          properties.push(entry);
        }

        // Inspect prototype methods without executing them.
        const prototypeMethods = [];
        let p = safe(() => Object.getPrototypeOf(obj));
        let depth = 0;

        while (p && depth < 4) {
          for (const key of propertyNames(p).slice(0, 150)) {
            if (sensitive.test(key)) continue;

            const d = descriptor(p, key);

            if (d && typeof d.value === 'function') {
              prototypeMethods.push({
                name: key,
                functionName: clean(d.value.name, 80)
              });
            }
          }

          p = safe(() => Object.getPrototypeOf(p));
          depth++;
        }

        return {
          path,
          constructor: safe(() => obj.constructor?.name),
          ownPropertyCount: properties.length,
          properties,
          functions,
          relevantChildren: childObjects,
          prototypeMethods: prototypeMethods.slice(0, 180)
        };
      }

      for (const name of selected) {
        const d = descriptor(cw, name);

        if (!d) continue;

        const value = Object.prototype.hasOwnProperty.call(d, 'value')
          ? d.value
          : undefined;

        report.chartGlobals.push({
          name,
          type: d.get || d.set
            ? 'accessor'
            : typeof value,
          details:
            value !== undefined &&
            value !== null &&
            (typeof value === 'object' ||
             typeof value === 'function') &&
            name !== 'window'
              ? inspectObject(
                  value,
                  'iframe.' + name
                )
              : undefined
        });
      }

      // --------------------------------------------
      // 4. Discover TradingView chart DOM
      // --------------------------------------------

      const selectors = [
        '.chart-widget',
        '.chart-container',
        '.chart-markup-table',
        '.chart-gui-wrapper',
        '.pane',
        '.price-axis-container',
        '.time-axis',
        '.chart-toolbar',
        '.chart-controls-bar',
        '[class*="series"]',
        '[class*="drawing"]',
        'canvas',
        'svg'
      ];

      const elements = new Set();

      for (const selector of selectors) {
        for (const el of safe(
          () => doc.querySelectorAll(selector),
          []
        )) {
          if (elements.size < 300) elements.add(el);
        }
      }

      for (const el of elements) {
        const item = info(el);
        const c = classes(el);
        const tag = el.tagName;

        if (tag === 'CANVAS') {
          report.dom.canvases.push({
            ...item,
            width: el.width,
            height: el.height
          });
        } else if (tag === 'SVG') {
          report.dom.svgs.push(item);
        }

        if (
          /chart-widget|chart-container|chart-markup-table|chart-gui-wrapper/i.test(c)
        ) {
          report.dom.chartContainers.push(item);
        }

        if (
          /\bpane\b|price-axis-container|time-axis/i.test(c)
        ) {
          report.dom.panes.push(item);
        }
      }

      // --------------------------------------------
      // 5. Timeframe and toolbar controls
      // --------------------------------------------

      const controls = safe(
        () => doc.querySelectorAll(
          'button,[role="button"],[role="tab"],[data-name]'
        ),
        []
      );

      for (const el of controls) {
        const item = info(el);

        if (report.dom.buttons.length < 160) {
          report.dom.buttons.push(item);
        }

        const combined = [
          item.text,
          item.aria,
          item.title,
          item.className,
          JSON.stringify(item.attributes)
        ].join(' ');

        if (
          /1m|3m|5m|15m|30m|1h|4h|1d|interval|resolution|timeframe|time frame/i.test(combined)
        ) {
          report.dom.timeframeControls.push(item);
        }
      }

      // --------------------------------------------
      // 6. Scripts loaded inside iframe
      // --------------------------------------------

      report.scripts = safe(
        () => Array.from(doc.scripts)
          .filter(s => s.src)
          .map(s => {
            try {
              const u = new URL(s.src, cw.location.href);
              return u.origin + u.pathname;
            } catch (_) {
              return '[unavailable]';
            }
          })
          .slice(0, 100),
        []
      );

      // --------------------------------------------
      // 7. Resource entries in iframe context
      // --------------------------------------------

      const entries = safe(
        () => cw.performance.getEntriesByType('resource'),
        []
      );

      const resourceMap = new Map();

      for (const entry of entries) {
        let url;

        try {
          const u = new URL(entry.name, cw.location.href);
          url = u.origin + u.pathname;
        } catch (_) {
          continue;
        }

        if (
          !/chart|tradingview|kline|datafeed|futures|study|socket/i.test(url)

        resourceMap.set(url, {
          url,
          type: entry.initiatorType,
          durationMs: Math.round(entry.duration),
          transferSize: entry.transferSize || 0
        });
      }

      report.chartResources = Array.from(
        resourceMap.values()
      ).slice(-100);

    } else {
      report.result.frameAccessError =
        'Iframe document is not accessible from the parent context';
    }
  }

  // ----------------------------------------------
  // 8. Summary
  // ----------------------------------------------

  report.finishedAt = new Date().toISOString();

  report.result = {
    ...report.result,

    iframeFound: !!iframe,

    iframeDocumentAccessible:
      !!safe(() => iframe.contentDocument),

    chartGlobalCount:
      report.chartGlobals.length,

    chartContainerCount:
      report.dom.chartContainers.length,

    canvasCount:
      report.dom.canvases.length,

    svgCount:
      report.dom.svgs.length,

    paneCount:
      report.dom.panes.length,

    timeframeControlCount:
      report.dom.timeframeControls.length,

    scriptCount:
      report.scripts.length,

    chartResourceCount:
      report.chartResources.length
  };

  // ----------------------------------------------

  // ----------------------------------------------

  const output = JSON.stringify(report, null, 2);

      noWebSocketCreated: true,
      noStorageRead: true,
      noCookiesOrTokensRead: true,
      sensitivePropertiesExcluded: true
    }
  };

  const safe = (fn, fallback = null) => {
    try {
      const value = fn();
      return value === undefined ? fallback : value;
    } catch (e) {
      return fallback;
    }
  };

  const sensitive =
    /token|password|secret|cookie|authorization|signature|api.?key|sessionid/i;

  function ownKeys(obj) {
    return safe(
      () => Object.getOwnPropertyNames(obj)
        .filter(k => !sensitive.test(k)),
      []
    );
  }

  function ownValue(obj, key) {
    const d = safe(
      () => Object.getOwnPropertyDescriptor(obj, key)
    );

    if (!d || !Object.prototype.hasOwnProperty.call(d, 'value')) {
      return undefined;
    }

    return d.value;
  }

  function findMethod(obj, name) {
    let p = obj;

    for (let i = 0; p && i < 8; i++) {
      const d = safe(
        () => Object.getOwnPropertyDescriptor(p, name)
      );

      if (d && typeof d.value === 'function') {
        return d.value;
      }

      p = safe(() => Object.getPrototypeOf(p));
    }

    return null;
  }

  function callRead(obj, name, args = []) {
    const fn = findMethod(obj, name);

    if (!fn) {
      return {
        found: false,
        reason: 'Method not found'
      };
    }

    try {
      return {
        found: true,
        ok: true,
        value: fn.apply(obj, args)
      };
    } catch (e) {
      return {
        found: true,
        ok: false,
        error: String(e?.message || e)
      };
    }
  }

  function simpleValue(value, depth = 0) {
    if (
      value === null ||
      typeof value === 'string' ||
      typeof value === 'number' ||
      typeof value === 'boolean'
    ) {
      return value;
    }

    if (value === undefined) return null;

    if (depth > 2) return '[Object]';

    if (Array.isArray(value)) {
      return value.slice(0, 25).map(v =>
        simpleValue(v, depth + 1)
      );
    }

    if (typeof value === 'function') {
      return '[Function]';
    }

    return '[Object]';
  }

  function inspect(obj, path, maxDepth = 0) {
    if (!obj || (
      typeof obj !== 'object' &&
      typeof obj !== 'function'
    )) {
      return {
        path,
        type: typeof obj
      };
    }

    const keys = ownKeys(obj);

    const result = {
      path,
      constructor: safe(
        () => obj.constructor?.name,
        null
      ),
      ownPropertyCount: keys.length,
      properties: [],
      methods: [],
      relevantChildren: []
    };

    for (const key of keys.slice(0, 180)) {
      const d = safe(
        () => Object.getOwnPropertyDescriptor(obj, key)
      );

      if (!d) continue;

      if (
        !Object.prototype.hasOwnProperty.call(d, 'value')
      ) {
        result.properties.push({
          name: key,
          type: 'accessor'
        });
        continue;
      }

      const value = d.value;

      const item = {
        name: key,
        type: typeof value,
        constructor: value !== null
          ? safe(() => value?.constructor?.name)
          : null
      };

      if (typeof value === 'function') {
        result.methods.push({
          name: key,
          functionName: value.name || '',
          argumentCount: value.length
        });
      }

      if (
        value !== null &&
        value !== undefined &&
        (typeof value === 'object' ||
         typeof value === 'function') &&
        /chart|series|pane|study|model|feed|scale|source|data|time|interval|resolution/i.test(key)
      ) {
        const child = {
          name: key,
          type: typeof value,
          constructor: item.constructor,
          keys: ownKeys(value).slice(0, 100)
        };

        if (maxDepth > 0 && typeof value === 'object') {
          child.details = inspect(
            value,
            path + '.' + key,
            maxDepth - 1
          );
        }

        result.relevantChildren.push(child);
      }

      if (
        value === null ||
        ['string', 'number', 'boolean'].includes(typeof value)
      ) {
        item.value = simpleValue(value);
      }

      result.properties.push(item);
    }

    // Inspect prototype methods without executing them.

    const protoMethods = [];
    let p = safe(() => Object.getPrototypeOf(obj));

    for (let depth = 0; p && depth < 4; depth++) {
      for (const key of safe(() => Object.getOwnPropertyNames(p), [])) {
        if (sensitive.test(key)) continue;

        const d = safe(
          () => Object.getOwnPropertyDescriptor(p, key)
        );

        if (d && typeof d.value === 'function') {
          protoMethods.push({
            name: key,
            argumentCount: d.value.length
          });
        }
      }

      p = safe(() => Object.getPrototypeOf(p));
    }

    result.prototypeMethods = protoMethods.slice(0, 180);

    return result;
  }

  function summarizeList(list, limit = 12) {
    if (!Array.isArray(list)) {
      return {
        type: typeof list,
        constructor: safe(() => list?.constructor?.name)
      };
    }

    return {
      count: list.length,
      items: list.slice(0, limit).map((item, index) => ({
        index,
        constructor: safe(() => item?.constructor?.name),
        keys: item && typeof item === 'object'
          ? ownKeys(item).slice(0, 100)
          : [],
        properties: item && typeof item === 'object'
          ? inspect(item, 'item_' + index).properties
              .filter(x =>
                ['string', 'number', 'boolean'].includes(x.type)
              )
              .slice(0, 30)
          : simpleValue(item)
      }))
    };
  }

  // 1. Locate chart iframe
  // -----------------------------------------

  const iframe =
    document.querySelector(
      '#futures-trade-tchart iframe'
    ) ||
    document.querySelector(
      'iframe[id^="tradingview_"]'
    );

  } else {

    const doc = safe(() => iframe.contentDocument);

        width: doc.documentElement.clientWidth,
        height: doc.documentElement.clientHeight
      };

      // ---------------------------------------
      // 2. Resolve known TradingView objects
      // ---------------------------------------

      const globals = {};

      for (const name of [

        'ChartApiInstance',

            )
          };
        }
      }

      // ---------------------------------------
      // 4. Public TradingView Chart API
      // ---------------------------------------

      let publicChart = null;

      if (tradingApi) {
        report.publicChartApi =
          inspect(tradingApi, 'tradingViewApi', 0);

        // These methods are read-only accessors.
        const active = callRead(tradingApi, 'activeChart');
        const chart = callRead(tradingApi, 'chart');

        report.publicChartApi.accessors = {

        };

        if (

        } else if (

      }

      // ---------------------------------------
      // 5. Public Chart read-only state
      // ---------------------------------------

      if (publicChart) {
        const getterNames = [
          'symbol',

          'getVisibleRange',

        ];

        const values = {};

        for (const name of getterNames) {

          if (

            name === 'getAllStudies'
          ) {

          }

          if (

          if (
            r.ok &&
            r.value &&
            typeof r.value === 'object' &&
            name === 'getTimeScale'
          ) {
            report.chartModel.timeScale =
              inspect(r.value, 'getTimeScale()', 0);
          }
        }

        report.publicChartApi.readOnlyValues = values;
      }

      // ---------------------------------------
      // 6. Chart Widget internals
      // ---------------------------------------

      let widget = directWidget;

      if (
        !widget &&
        collection
      ) {
        const all = callRead(collection, 'getAll');

        if (all.ok && Array.isArray(all.value)) {
          widget = all.value[0] || null;
        }
      }

      if (widget) {
        report.chartWidget = inspect(
          widget,
          'chartWidget',
          1
        );

        const widgetGetters = [
          'getSymbol',
          'getResolution',
          'model',
          'getTimeScale',
          'paneWidgets'
        ];

        report.chartWidget.readOnlyGetters = {};

        for (const name of widgetGetters) {
          const r = callRead(widget, name);

          report.chartWidget.readOnlyGetters[name] = {
            found: r.found,
            ok: r.ok,
            type: r.ok ? typeof r.value : null,
            value: r.ok
              ? simpleValue(r.value)
              : null,
            error: r.error || null
          };

          if (
            r.ok &&
            r.value &&
            typeof r.value === 'object' &&
            name === 'model'
          ) {
            report.chartModel.model =
              inspect(r.value, 'chartWidget.model()', 1);

            const model = r.value;

            const mainSeries = callRead(
              model,
              'mainSeries'
            );

            if (
              mainSeries.ok &&
              mainSeries.value
            ) {
              report.series.mainSeries =
                inspect(
                  mainSeries.value,
                  'model.mainSeries()',
                  1
                );
            }

            const dataSources = callRead(
              model,
              'dataSources'
            );

            if (dataSources.ok) {
              report.chartModel.dataSources =
                summarizeList(dataSources.value, 25);
            }
          }

          if (
            r.ok &&
            Array.isArray(r.value) &&
            name === 'paneWidgets'
          ) {
            report.panes.push(
              ...summarizeList(r.value, 15).items
            );
          }
        }

        // Current symbol and resolution WatchedValues
        for (const key of [
          '_symbolWV',
          '_resolutionWV'
        ]) {
          const watched = ownValue(widget, key);

          if (watched) {
            report.chartWidget[key] = {
              constructor: safe(
                () => watched.constructor?.name
              ),
              value: simpleValue(
                ownValue(watched, '_value')
              ),
              keys: ownKeys(watched)
            };
          }
        }

        // Pane widgets without invoking their methods
        const paneWidgets = ownValue(
          widget,
          '_paneWidgets'
        );

        if (Array.isArray(paneWidgets)) {
          report.chartWidget.paneWidgetDetails =
            paneWidgets.slice(0, 12).map(
              (pane, i) =>
                inspect(pane, 'paneWidget_' + i, 0)
            );
        }

      // 7. ChartApiInstance / DataFeed
      // ---------------------------------------

        );

            '_datafeedConfiguration',

    return safe(() =>
      Object.getOwnPropertyNames(obj)
        .filter(k => !sensitive.test(k))
    , []);
  }

  function ownValue(obj, key) {
    const d = safe(() =>
      Object.getOwnPropertyDescriptor(obj, key)
    );

    if (
      !d ||
      !Object.prototype.hasOwnProperty.call(d, 'value')
    ) return undefined;

    return d.value;
  }

  function findMethod(obj, name) {
    let p = obj;

    for (let i = 0; p && i < 8; i++) {
      const d = safe(() =>
        Object.getOwnPropertyDescriptor(p, name)
      );

      if (d && typeof d.value === 'function') {
        return d.value;
      }

      p = safe(() => Object.getPrototypeOf(p));
    }

    return null;
  }

  function readMethod(obj, name) {
    const fn = findMethod(obj, name);

    if (!fn) {
      return {
        found: false,
        ok: false
      };
    }

    try {
      const value = fn.call(obj);

      return {
        found: true,
        ok: true,
        value
      };
    } catch (e) {
      report.errors.push(
        name + ': ' + String(e?.message || e)
      );

      return {
        found: true,
        ok: false,
        error: String(e?.message || e)
      };
    }
  }

  function inspect(obj, path, maxMethods = 150) {
    if (
      !obj ||
      (typeof obj !== 'object' &&
       typeof obj !== 'function')
    ) {
      return {
        path,
        type: typeof obj
      };
    }

    const keys = ownKeys(obj);

    const result = {
      path,
      constructor: safe(
        () => obj.constructor?.name
      ),
      ownPropertyCount: keys.length,
      properties: [],
      methods: [],
      relevantChildren: [],
      prototypeMethods: []
    };

    for (const key of keys.slice(0, 180)) {
      const d = safe(() =>
        Object.getOwnPropertyDescriptor(obj, key)
      );

      if (!d) continue;

      if (!Object.prototype.hasOwnProperty.call(d, 'value')) {
        result.properties.push({
          name: key,
          type: 'accessor'
        });
        continue;
      }

      const value = d.value;

      const item = {
        name: key,
        type: typeof value,
        constructor: value == null
          ? null
          : safe(() => value.constructor?.name)
      };

      if (typeof value === 'function') {
        result.methods.push({
          name: key,
          functionName: value.name || '',
          argumentCount: value.length
        });
      }

      if (
        value === null ||
        ['string', 'number', 'boolean'].includes(typeof value)
      ) {
        item.value = value;
      }

      if (
        value !== null &&
        value !== undefined &&
        (typeof value === 'object' ||
         typeof value === 'function') &&
        /chart|series|pane|study|model|feed|scale|source|data|time|interval|resolution/i.test(key)
      ) {
        result.relevantChildren.push({
          name: key,
          type: typeof value,
          constructor: item.constructor,
          keys: ownKeys(value).slice(0, 100)
        });
      }

      result.properties.push(item);
    }

    let p = safe(() => Object.getPrototypeOf(obj));
    let depth = 0;

    while (p && depth < 4) {
      for (const key of safe(
        () => Object.getOwnPropertyNames(p),
        []
      )) {
        if (sensitive.test(key)) continue;

        const d = safe(() =>
          Object.getOwnPropertyDescriptor(p, key)
        );

        if (d && typeof d.value === 'function') {
          result.prototypeMethods.push({
            name: key,
            argumentCount: d.value.length
          });
        }
      }

      p = safe(() => Object.getPrototypeOf(p));
      depth++;
    }

    result.methods = result.methods.slice(0, maxMethods);
    result.prototypeMethods =
      result.prototypeMethods.slice(0, maxMethods);

    return result;
  }

  function summarizeList(list, limit = 15) {
    if (!Array.isArray(list)) {
      return {
        type: typeof list,
        constructor: safe(() => list?.constructor?.name)
      };
    }

    return {
      count: list.length,
      items: list.slice(0, limit).map((item, index) => ({
        index,
        constructor: safe(() => item?.constructor?.name),
        keys: item && typeof item === 'object'
          ? ownKeys(item).slice(0, 100)
          : [],
        details: item && typeof item === 'object'
          ? inspect(item, 'item_' + index, 80)
          : item
      }))
    };
  }

  function readSimple(obj, methods) {
    const out = {};

    for (const name of methods) {
      const r = readMethod(obj, name);

      out[name] = {
        found: r.found,
        ok: r.ok,
        value: r.ok && (
          r.value === null ||
          ['string', 'number', 'boolean'].includes(typeof r.value)
        )
          ? r.value
          : null,
        constructor: r.ok
          ? safe(() => r.value?.constructor?.name)
          : null,
        error: r.error || null
      };
    }

    return out;
  }

  // 1. Find the actual chart iframe
  // ----------------------------------------

  const iframe =
    document.querySelector('#futures-trade-tchart iframe') ||
    document.querySelector('iframe[id^="tradingview_"]');

  let doc = null;

    }

    // --------------------------------------
    // 7. Chart Widget model
    // --------------------------------------

    if (chartWidget) {
      report.chartModel.widget =

      const modelResult =
        readMethod(chartWidget, 'model');

      if (modelResult.ok && modelResult.value) {

        report.chartModel.model =
          inspect(model, 'chartWidget.model()', 100);

        // Main series
        const mainSeriesResult =
          readMethod(model, 'mainSeries');

        if (mainSeriesResult.ok && mainSeriesResult.value) {

              mainSeriesResult.value,

              100
            );
        }

        // Data sources
        const dataSources =
          readMethod(model, 'dataSources');

        if (dataSources.ok) {

        }
      }

      // Current symbol and resolution WatchedValues
      for (const key of [
        '_symbolWV',
        '_resolutionWV'
      ]) {
        const watched = ownValue(chartWidget, key);

        if (watched) {
          report.chartModel[key] = {
            constructor: safe(
              () => watched.constructor?.name
            ),
            value: ownValue(watched, '_value'),
            keys: ownKeys(watched)
          };
        }
      }

      // Pane widget structures
      const paneWidgets =
        ownValue(chartWidget, '_paneWidgets');

      if (Array.isArray(paneWidgets)) {
        report.chartModel.paneWidgets =
          paneWidgets.slice(0, 12).map(
            (pane, i) => inspect(
              pane,
              'paneWidget_' + i,
              70

          );
      }
    }

    // --------------------------------------
    // 8. DataFeed engine
    // --------------------------------------

    if (chartApiInstance) {
      report.chartApi =
        inspect(chartApiInstance, 'ChartApiInstance', 120);

      studyEngine =
        ownValue(chartApiInstance, '_studyEngine');

      if (studyEngine) {
        report.datafeed.studyEngine =
          inspect(studyEngine, '_studyEngine', 120);

        for (const key of [
          '_externalDatafeed',
          '_datafeedConfiguration',
          '_cachedDatafeed',
          '_resolveRequests',
          '_resolvePromisesBySymbolId',
          '_symbolIdToSymbolRequestString',
          '_timeScales',
          '_barsCoefficientsCache'
        ]) {
          const value =
            ownValue(studyEngine, key);

          if (value !== undefined) {
            report.datafeed[key] = {
              type: typeof value,
              constructor: value == null
                ? null
                : safe(() => value.constructor?.name),
              keys: value &&
                (typeof value === 'object' ||
                 typeof value === 'function')
                ? ownKeys(value).slice(0, 100)
                : [],
              details: value &&
                (typeof value === 'object' ||
                 typeof value === 'function')

        }

        const externalDatafeed =

        report.datafeed.externalDatafeedFound =

    errors: [],
    iframe: {},
    chart: {},
    series: {},
    data: {},

  const safe = (fn, fallback = null) => {
    try {
      const v = fn();
      return v === undefined ? fallback : v;
    } catch (e) {
      report.errors.push(String(e?.message || e));
      return fallback;
    }
  };

  const sensitive =

    return safe(() =>
      Object.getOwnPropertyNames(obj)
        .filter(k => !sensitive.test(k))
    , []);
  }

  function value(obj, key) {
    const d = safe(() =>
      Object.getOwnPropertyDescriptor(obj, key)
    );

    return d &&
      Object.prototype.hasOwnProperty.call(d, 'value')
        ? d.value
        : undefined;
  }

  function method(obj, name) {
    let p = obj;

    for (let i = 0; p && i < 8; i++) {
      const d = safe(() =>
        Object.getOwnPropertyDescriptor(p, name)
      );

      if (d && typeof d.value === 'function') {
        return d.value;
      }

      p = safe(() => Object.getPrototypeOf(p));
    }

    return null;
  }

    const fn = method(obj, name);

    if (!fn) {
      return { found: false, ok: false };

    try {
      return {
        found: true,
        ok: true,
        value: fn.call(obj)
      };
    } catch (e) {
      report.errors.push(
        name + ': ' + String(e?.message || e)
      );

      return {
        found: true,
        ok: false,
        error: String(e?.message || e)
      };
    }
  }

  function describe(obj, path) {
    if (!obj || typeof obj !== 'object') {
      return {
        path,
        type: typeof obj
      };
    }

    const own = keys(obj);

    let proto = safe(() => Object.getPrototypeOf(obj));
    const protoMethods = [];
    let depth = 0;

    while (proto && depth < 4) {
      for (const name of keys(proto)) {
        const d = safe(() =>
          Object.getOwnPropertyDescriptor(proto, name)
        );

        if (d && typeof d.value === 'function') {
          protoMethods.push({
            name,
            arguments: d.value.length
          });
        }
      }

      proto = safe(() => Object.getPrototypeOf(proto));
      depth++;
    }

    const properties = [];

    for (const name of own.slice(0, 130)) {
      const d = safe(() =>
        Object.getOwnPropertyDescriptor(obj, name)
      );

      if (!d) continue;

      if (!Object.prototype.hasOwnProperty.call(d, 'value')) {
        properties.push({
          name,
          type: 'accessor'
        });
        continue;
      }

      const v = d.value;

      const item = {
        name,
        type: typeof v,
        constructor: v == null
          ? null
          : safe(() => v.constructor?.name)
      };

      if (
        v === null ||
        ['string', 'number', 'boolean'].includes(typeof v)
      ) {
        item.value = v;
      }

      if (typeof v === 'function') {
        item.arguments = v.length;
      }

      if (
        v &&
        typeof v === 'object' &&
        /data|bar|plot|builder|series|resolution|interval|symbol/i.test(name)
      ) {
        item.keys = keys(v).slice(0, 60);
      }

      properties.push(item);
    }

    return {
      path,
      constructor: safe(() => obj.constructor?.name),
      ownPropertyCount: own.length,
      properties,
      prototypeMethods: protoMethods
    };
  }

        : null;

      };

      // 4. Read Series Data
      // -------------------------------------

      const dataResult =

      const data =

      // -------------------------------------

          'getBars',

          'unsubscribeBars',

      // -------------------------------------

  ```text
    volume,
    extra
  ]
  ```

  <text color="success" size="sm" weight="medium">داده واقعی و قابل خواندن از Bar Store استخراج شد.</text>
</box>

### ۲. ساختار ذخیره‌سازی کندل‌ها

این اجزا پیدا شدند:

- `mainSeries.data()`
- `m_bars`
- `m_nsBars`
- `_items`
- متد `merge()` در Bar Store
- متدهای `addTail()` و `move()`

یعنی حالا آرایه واقعی کندل‌ها و ساختار نگهداری آن‌ها را داریم.

### ۳. چیزی که هنوز پیدا نشده

<badge color="warning">Builder ناشناخته</badge>

در KCEX از ساختاری به نام `_conflatedChunksBuilder` استفاده کرده بودیم. در Bitunix این property در محل بررسی‌شده وجود ندارد.

بنابراین نباید فعلاً همان روش قبلی را عیناً اجرا کنیم.

### ۴. وضعیت تایم‌فریم ثانیه‌ای

در تنظیمات فعلی DataFeed این ۲۳ مقدار ثبت شده‌اند:

`1, 2, 3, 5, 6, 9, 10, 12, 15, 24, 27, 30, 45, 60, 120, 240, 360, 480, 720, 1D, 3D, 1W, 1M`

این‌ها Resolutionهای پشتیبانی‌شده‌ای هستند که DataFeed گزارش کرده است. هنوز ثانیه‌ای بودن آن‌ها تأیید نشده؛ به‌ویژه کندل‌های نمونه با فاصله ۶۰ ثانیه ثبت شده‌اند.

## مرحله هفتم: شناسایی موتور به‌روزرسانی کندل‌ها

در این مرحله می‌خواهم بفهمم کندل زنده چطور به Bar Store اضافه یا اصلاح می‌شود.

این موضوع برای جلوگیری از خراب‌شدن اندیکاتورها هنگام فعال‌کردن Align Candle Opens بسیار مهم است.

کد شماره ۷ این موارد را استخراج می‌کند:

- ساختار `_pressedChunks`
- ارتباط `SeriesSource` با Bar Store
- ساختار `_gateway` و متدهای آن
- رویدادهای `_barReceived` و `_dataUpdated`
- ساختار کندل‌های تاریخی و زنده
- مشخصات دقیق Resolution و Symbol
- متدهای DataFeed بدون اجرای آن‌ها

<badge color="success">کد شماره ۷ — فقط خواندن اطلاعات</badge>

<CodeBlock language="javascript" editable>
(() => {
  'use strict';

  const TAG = 'BITUNIX_LIVE_BAR_ENGINE_07';
  const PANEL = '__bitunix_live_bar_engine_07__';

  document.getElementById(PANEL)?.remove();

  const report = {
    diagnostic: TAG,
    startedAt: new Date().toISOString(),
    errors: [],
    symbol: {},
    series: {},
    interval: {},
    barStore: {},
    pressedChunks: {},
    seriesSource: {},
    gateway: {},
    dataEvents: {},
    datafeedMethods: {},
    result: {},
    safety: {
      readOnly: true,
      noOrdersSent: true,
      noClicks: true,
      noCandleChanges: true,
      noSubscriptions: true,
      noDatafeedCalls: true,
      noNetworkRequests: true,
      noWebSocketsCreated: true,
      noStorageOrCookiesRead: true
    }
  };

  const safe = (fn, fallback = null) => {
    try {
      const v = fn();
      return v === undefined ? fallback : v;
    } catch (e) {
      report.errors.push(String(e?.message || e));
      return fallback;
    }
  };

  const sensitive =
    /token|password|secret|cookie|authorization|signature|api.?key|sessionid/i;

  function keys(obj) {
    return safe(() =>
      Object.getOwnPropertyNames(obj)
        .filter(k => !sensitive.test(k))
    , []);
  }

  function val(obj, key) {
    const d = safe(() =>
      Object.getOwnPropertyDescriptor(obj, key)
    );

    if (
      !d ||
      !Object.prototype.hasOwnProperty.call(d, 'value')
    ) return undefined;

    return d.value;
  }

  function findMethod(obj, name) {
    let p = obj;

    for (let i = 0; p && i < 8; i++) {
      const d = safe(() =>
        Object.getOwnPropertyDescriptor(p, name)
      );

      if (d && typeof d.value === 'function') {
        return d.value;
      }

      p = safe(() => Object.getPrototypeOf(p));
    }

    return null;
  }

  function callRead(obj, name) {
    const fn = findMethod(obj, name);

    if (!fn) return {
      found: false,
      ok: false
    };

    try {
      return {
        found: true,
        ok: true,
        value: fn.call(obj)
      };
    } catch (e) {
      report.errors.push(name + ': ' + String(e?.message || e));

      return {
        found: true,
        ok: false,
        error: String(e?.message || e)
      };
    }
  }

  function methods(obj) {
    const result = [];
    let p = obj;

    for (let depth = 0; p && depth < 5; depth++) {
      for (const name of keys(p)) {
        const d = safe(() =>
          Object.getOwnPropertyDescriptor(p, name)
        );

        if (d && typeof d.value === 'function') {
          result.push({
            name,
            arguments: d.value.length
          });
        }
      }

      p = safe(() => Object.getPrototypeOf(p));
    }

    return result.slice(0, 130);
  }

  function structure(obj, path, max = 100) {
    if (!obj || typeof obj !== 'object') {
      return {
        path,
        type: typeof obj
      };
    }

    const out = {
      path,
      constructor: safe(() => obj.constructor?.name),
      keys: keys(obj).slice(0, max),
      primitiveProperties: [],
      methods: methods(obj)
    };

    for (const key of keys(obj).slice(0, max)) {
      const v = val(obj, key);

      if (
        v === null ||
        ['string', 'number', 'boolean'].includes(typeof v)
      ) {
        out.primitiveProperties.push({
          name: key,
          value: v
        });
      }
    }

    return out;
  }

  function sampleRows(store, count = 5) {
    const items = val(store, '_items');

    if (!Array.isArray(items)) {
      return {
        found: false,
        reason: '_items is not an array'
      };
    }

    return {
      count: items.length,
      first: items.slice(0, count).map((row, i) => ({
        position: i,
        index: val(row, 'index') ?? null,
        value: Array.isArray(val(row, 'value'))
          ? val(row, 'value').slice(0, 7)
          : null
      })),
      last: items.slice(-count).map((row, i) => ({
        position: items.length - count + i,
        index: val(row, 'index') ?? null,
        value: Array.isArray(val(row, 'value'))
          ? val(row, 'value').slice(0, 7)
          : null
      }))
    };
  }

  // 1. Find TradingView iframe

  const iframe =
    document.querySelector('#futures-trade-tchart iframe') ||
    document.querySelector('iframe[id^="tradingview_"]');

  const win = safe(() => iframe?.contentWindow);
  const doc = safe(() => iframe?.contentDocument);

  if (!win || !doc) {
    report.errors.push('TradingView iframe inaccessible');
  } else {

    report.iframe = {
      id: iframe.id,
      readyState: doc.readyState
    };

    // 2. Resolve chart objects

    const chartApi = val(win, 'ChartApiInstance');
    const widget = val(win, 'chartWidget');

    const collection = val(win, 'chartWidgetCollection');

    const model = widget
      ? callRead(widget, 'model').value
      : null;

    const series = model
      ? callRead(model, 'mainSeries').value
      : null;

    if (!series) {
      report.errors.push('Main Series not found');
    } else {

      // 3. Series identity

      report.series = {
        symbol: safe(
          () => callRead(series, 'symbol').value
        ),
        interval: safe(
          () => callRead(series, 'interval').value
        ),
        intervalObject: structure(
          val(series, '_intervalObj'),
          'series._intervalObj'
        ),
        symbolInfo: {
          properties: structure(
            val(series, '_symbolInfo'),
            'series._symbolInfo'
          ),
          supportedResolutions: safe(() =>
            val(val(series, '_symbolInfo'), 'supported_resolutions')
          )
        }
      };

      // 4. Series data and Bar Store

      const data = callRead(series, 'data').value;
      const bars = data
        ? callRead(data, 'bars').value
        : null;

      const nsBars = data
        ? callRead(data, 'nsBars').value
        : null;

      report.barStore = {
        dataFound: !!data,
        dataStructure: structure(
          data,
          'mainSeries.data()'
        ),
        barsStructure: structure(
          bars,
          'data.bars()'
        ),
        barsSamples: sampleRows(bars, 5),
        nsBarsStructure: structure(
          nsBars,
          'data.nsBars()'
        ),
        nsBarsSamples: sampleRows(nsBars, 3),
        lastIndex: safe(() =>
          callRead(bars, 'lastIndex').value
        ),
        firstIndex: safe(() =>
          callRead(bars, 'firstIndex').value
        )
      };

      // 5. Pressed Chunks

      const pressedChunks = data
        ? val(data, '_pressedChunks')
        : null;

      report.pressedChunks = {
        found: !!pressedChunks,
        structure: structure(
          pressedChunks,
          'data._pressedChunks'
        ),
        entries: []
      };

      if (
        pressedChunks &&
        typeof pressedChunks === 'object'
      ) {
        for (const key of keys(pressedChunks).slice(0, 25)) {
          const item = val(pressedChunks, key);

          report.pressedChunks.entries.push({
            name: key,
            constructor: safe(
              () => item?.constructor?.name
            ),
            keys: item && typeof item === 'object'
              ? keys(item).slice(0, 35)
              : [],
            length: Array.isArray(item)
              ? item.length
              : null
          });
        }
      }

      // 6. Series Source

      const source =
        callRead(series, 'seriesSource').value ||
        val(series, '_seriesSource');

      report.seriesSource = {
        found: !!source,
        structure: structure(
          source,
          'mainSeries._seriesSource'
        )
      };

      if (source) {
        for (const name of [
          '_resolution',
          '_timeFrame',
          '_status',
          '_turnaroundCounter',
          '_ongoingDataUpdate',
          '_createSeriesParams',
          '_extSymbol'
        ]) {
          const v = val(source, name);

          if (
            v === null ||
            ['string', 'number', 'boolean'].includes(typeof v)
          ) {
            report.seriesSource[name] = v;
          } else if (v && typeof v === 'object') {
            report.seriesSource[name + '_structure'] =
              structure(v, 'seriesSource.' + name, 60);
          }
        }

        // 7. Gateway

        const gateway = val(source, '_gateway');

        report.gateway = {
          found: !!gateway,
          structure: structure(
            gateway,
            'seriesSource._gateway'
          )
        };

        // 8. Data Events

        const events =
          val(source, '_dataEvents') ||
          callRead(source, 'dataEvents').value;

        report.dataEvents = {
          found: !!events,
          structure: structure(
            events,
            'seriesSource._dataEvents'
          ),
          eventDetails: []
        };

        if (events) {
          for (const name of keys(events).slice(0, 70)) {
            const event = val(events, name);

            if (!event || typeof event !== 'object') continue;

            const listeners = val(event, '_listeners');

            report.dataEvents.eventDetails.push({
              name,
              constructor: safe(
                () => event.constructor?.name
              ),
              keys: keys(event).slice(0, 20),
              listenerCount: Array.isArray(listeners)
                ? listeners.length
                : listeners && typeof listeners === 'object'
                  ? keys(listeners).length
                  : null
            });
          }
        }
      }

      // 9. DataFeed method signatures

      const engine = chartApi
        ? val(chartApi, '_studyEngine')
        : null;

      const feed = engine
        ? val(engine, '_externalDatafeed')
        : null;

      if (feed) {
        const names = [
          'onReady',
          'resolveSymbol',
          'getBars',
          'subscribeBars',
          'unsubscribeBars',
          'getServerTime',
          'searchSymbols'
        ];

        report.datafeedMethods = {
          found: true,
          methods: names.map(name => {
            const fn = findMethod(feed, name);

            return {
              name,
              found: !!fn,
              argumentCount: fn?.length ?? null,
              functionType: fn?.constructor?.name ?? null
            };
          })
        };
      }

      // 10. Final result

      report.result = {
        symbol: report.series.symbol,
        resolution: report.series.interval,
        barsCount: report.barStore.barsSamples.count,
        nsBarsFound: !!nsBars,
        pressedChunksFound: !!pressedChunks,
        seriesSourceFound: !!source,
        gatewayFound: !!val(source, '_gateway'),
        dataEventsFound: !!val(source, '_dataEvents'),
        liveBarReceivedEventFound:
          !!val(
            val(source, '_dataEvents'),
            '_barReceived'
          ),
        liveDataUpdatedEventFound:
          !!val(
            val(source, '_dataEvents'),
            '_dataUpdated'
          ),
        externalDatafeedFound: !!feed
      };
    }
  }

  report.finishedAt = new Date().toISOString();

  // 11. Report panel

  const output = JSON.stringify(report, null, 2);

  const overlay = document.createElement('div');
  overlay.id = PANEL;

  overlay.style.cssText = `
    position:fixed;
    inset:0;
    z-index:2147483647;
    background:rgba(3,8,20,.84);
    backdrop-filter:blur(14px);
    display:flex;
    align-items:center;
    justify-content:center;
    padding:16px;
    box-sizing:border-box;
    color:#e5efff;
    font-family:Arial,sans-serif;
  `;

  const box = document.createElement('div');

  box.style.cssText = `
    width:min(1000px,96vw);
    height:min(820px,94vh);
    background:rgba(16,25,44,.97);
    border:1px solid rgba(110,155,230,.35);
    border-radius:16px;
    padding:18px;
    display:flex;
    flex-direction:column;
    gap:12px;
    box-sizing:border-box;
  `;

  const title = document.createElement('div');
  title.textContent = TAG;
  title.style.cssText =
    'font-size:15px;font-weight:bold;color:#8fbdff;';

  const area = document.createElement('textarea');
  area.value = output;
  area.readOnly = true;

  area.style.cssText = `
    flex:1;
    width:100%;
    min-height:0;
    resize:none;
    box-sizing:border-box;
    background:rgba(4,9,20,.85);
    color:#dce9ff;
    border:1px solid rgba(120,160,220,.25);
    border-radius:9px;
    padding:12px;
    outline:none;
    font:11px/1.5 Consolas,monospace;
  `;

  const footer = document.createElement('div');
  footer.style.cssText =
    'display:flex;align-items:center;gap:12px;';

  const copy = document.createElement('button');
  copy.textContent = 'کپی گزارش کامل';

    border:0;
    border-radius:8px;

  `;

  const close = document.createElement('button');
  close.textContent = 'بستن';
  close.style.cssText = `

    color:white;
    border:1px solid rgba(255,255,255,.15);

  `;

  const status = document.createElement('span');
  status.textContent = 'آماده';
  status.style.cssText =
    'font-size:12px;color:#aab8d2;';

  copy.onclick = async () => {

  footer.append(copy, close, status);
  box.append(title, area, footer);
  overlay.append(box);
  document.documentElement.append(overlay);

  console.info('[BITUNIX SCAN 07] Read-only completed.');
