      // Keep only protocol host and path. Query parameters contain user/trace IDs.
      return `${u.protocol}//${u.host}${u.pathname}`;

  function sanitizeOutgoingText(text) {
    if (typeof text !== 'string') return text;
    try {
      const obj = JSON.parse(text);
      const sensitive = /token|sign|secret|userid|user_id|uid|apikey|api_key|traceid|trace_id|session|auth/i;
      const walk = (value) => {
        if (Array.isArray(value)) return value.map(walk);

          const out = {};
          for (const [k, v] of Object.entries(value)) out[k] = sensitive.test(k) ? '[REDACTED]' : walk(v);

    }

    let binary = '';

    for (let i = 0; i < bytes.length; i += step) {
      binary += String.fromCharCode(...bytes.subarray(i, Math.min(i + step, bytes.length)));
    }
    return btoa(binary);
  }

هدف

این افزونه فقط برای تشخیص ساختار پیام‌های WebSocket بازار Bitunix ساخته شده است. پیام‌های کانال /ws-tide-batch/ را در همان تب XAUTUSDT ضبط می‌کند تا بفهمیم آیا پیام اولیه شامل معاملات قبل از زمان اتصال هست یا فقط داده زنده را می‌فرستد.

- هیچ سفارش یا معامله‌ای ارسال نمی‌کند.
